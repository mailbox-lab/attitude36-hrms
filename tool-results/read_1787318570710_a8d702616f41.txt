   176→// ===== Kanban Card =====

   178→function KanbanCard({
   179→  job,
   180→  onView,
   181→  onEdit,
   182→  index,
   183→}: {
   184→  job: Job
   185→  onView: () => void
   186→  onEdit: () => void
   187→  index: number
   188→}) {
   189→  const currencySymbol = CURRENCY_SYMBOLS[job.currency] || job.currency
   190→  const hasSalary = job.salaryMin != null || job.salaryMax != null
   191→
   192→  return (
   193→    <motion.div
   194→      variants={cardVariants}
   195→      initial="hidden"
   196→      animate="visible"
   197→      custom={index}
   198→      className="group relative cursor-pointer rounded-lg border border-border/60 bg-card p-3.5 shadow-sm transition-all hover:-translate-y-0.5 hover:shadow-md"
   199→      onClick={onView}
   200→    >
   201→      {/* Actions menu */}
   202→      <div className="absolute right-2 top-2 z-10 opacity-0 transition-opacity group-hover:opacity-100">
   203→        <DropdownMenu>
   204→          <DropdownMenuTrigger asChild>
   205→            <Button
   206→              variant="ghost"
   207→              size="icon"
   208→              className="h-6 w-6 shrink-0"
   209→              onClick={(e) => e.stopPropagation()}
   210→            >
   211→              <MoreHorizontal className="h-3.5 w-3.5" />
   212→            </Button>
   213→          </DropdownMenuTrigger>
   214→          <DropdownMenuContent align="end">
   215→            <DropdownMenuItem
   216→              onClick={(e) => {
   217→                e.stopPropagation()
   218→                onView()
   219→              }}
   220→            >
   221→              <Eye className="mr-2 h-3.5 w-3.5" />
   222→              View Details
   223→            </DropdownMenuItem>
   224→            <DropdownMenuItem
   225→              onClick={(e) => {
   226→                e.stopPropagation()
   227→                onEdit()
   228→              }}
   229→            >
   230→              <Pencil className="mr-2 h-3.5 w-3.5" />
   231→              Edit
   232→            </DropdownMenuItem>
   233→          </DropdownMenuContent>
   234→        </DropdownMenu>
   235→      </div>
   236→
   237→      {/* Title */}
   238→      <p className="truncate pr-6 text-sm font-medium leading-tight">{job.title}</p>
   239→
   240→      {/* Client */}
   241→      <p className="mt-0.5 truncate text-xs text-muted-foreground">{job.client.name}</p>
   242→
   243→      {/* Location + Type row */}
   244→      <div className="mt-2.5 flex items-center gap-3 text-[10px] text-muted-foreground">
   245→        {job.location && (
   246→          <span className="flex items-center gap-0.5">
   247→            <MapPin className="h-2.5 w-2.5" /> {job.location}
   248→          </span>
   249→        )}
   250→        <span className="flex items-center gap-0.5">
   251→          <Clock className="h-2.5 w-2.5" /> {job.employmentType}
   252→        </span>
   253→      </div>
   254→
   255→      {/* Priority + Candidates row */}
   256→      <div className="mt-2.5 flex items-center justify-between">
   257→        <div className="flex items-center gap-1.5">
   258→          <span
   259→            className={`h-2 w-2 rounded-full ${PRIORITY_DOT_COLORS[job.priority] || 'bg-gray-400'}`}
   260→          />
   261→          <Badge
   262→            className={`text-[10px] font-medium ${PRIORITY_COLORS[job.priority] || ''}`}
   263→          >
   264→            {job.priority}
   265→          </Badge>
   266→          {job.openings > 1 && (
   267→            <span className="text-[10px] text-muted-foreground">
   268→              ({job.openings} pos)
   269→            </span>
   270→          )}
   271→        </div>
   272→        <span className="flex items-center gap-1 text-[10px] text-muted-foreground">
   273→          <Users className="h-3 w-3" />
   274→          {job._count.candidates} candidate{job._count.candidates !== 1 ? 's' : ''}
   275→        </span>
   276→      </div>
   277→
   278→      {/* Salary range */}
   279→      {hasSalary && (
   280→        <p className="mt-2 text-[10px] text-muted-foreground">
   281→          {currencySymbol}
   282→          {(job.salaryMin ?? 0).toLocaleString('en-IN')}
   283→          {job.salaryMax
   284→            ? ` – ${currencySymbol}${job.salaryMax.toLocaleString('en-IN')}`
   285→            : '+'}
   286→        </p>
   287→      )}
   288→
   289→      {/* Recruiter */}
   290→      {job.recruiter && (
   291→        <div className="mt-2 flex items-center gap-1 border-t border-border/50 pt-2 text-[10px] text-muted-foreground">
   292→          <User className="h-2.5 w-2.5" />
   293→          {job.recruiter.name}
   294→        </div>
   295→      )}
   296→    </motion.div>
   297→  )
   298→}

   300→// ===== Kanban Column =====
   301→
   302→function KanbanColumn({
   303→  status,
   304→  jobs,
   305→  onViewJob,
   306→  onEditJob,
   307→  columnIndex,
   308→}: {
   309→  status: string
   310→  jobs: Job[]
   311→  onViewJob: (id: string) => void
   312→  onEditJob: (job: Job) => void
   313→  columnIndex: number
   314→}) {
   315→  return (
   316→    <motion.div
   317→      variants={columnVariants}
   318→      initial="hidden"
   319→      animate="visible"
   320→      custom={columnIndex}
   321→      className="flex w-72 shrink-0 flex-col"
   322→    >
   323→      {/* Column header */}
   324→      <div className="mb-3 flex items-center justify-between px-1">
   325→        <div className="flex items-center gap-2">
   326→          <span
   327→            className={`h-2.5 w-2.5 rounded-full ${STATUS_DOT_COLORS[status] || 'bg-gray-400'}`}
   328→          />
   329→          <span className="text-sm font-semibold">{status}</span>
   330→        </div>
   331→        <span
   332→          className={`flex h-5 min-w-5 items-center justify-center rounded-full px-1.5 text-[10px] font-bold ${KANBAN_COUNT_COLORS[status] || 'bg-gray-500/15 text-gray-700'}`}
   333→        >
   334→          {jobs.length}
   335→        </span>
   336→      </div>
   337→
   338→      {/* Column body with gradient top border */}
   339→      <div
   340→        className="flex max-h-[calc(100vh-320px)] min-h-[120px] flex-col gap-2.5 overflow-y-auto rounded-lg rounded-t-none bg-muted/30 p-2.5"
   341→        style={{
   342→          borderTop: `3px solid ${KANBAN_BORDER_COLOR[status] || '#9ca3af'}`,
   343→        }}
   344→      >
   345→        {jobs.length === 0 ? (
   346→          <div className="flex flex-1 items-center justify-center py-8">
   347→            <p className="text-xs text-muted-foreground/60">No jobs</p>
   348→          </div>
   349→        ) : (
   350→          jobs.map((job, i) => (
   351→            <KanbanCard
   352→              key={job.id}
   353→              job={job}
   354→              onView={() => onViewJob(job.id)}
   355→              onEdit={() => onEditJob(job)}
   356→              index={i}
   357→            />
   358→          ))
   359→        )}
   360→      </div>
   361→    </motion.div>
   362→  )
   363→}

   367→export function JobsPage() {
   368→  const navigate = useCRMStore((s) => s.navigate)
   369→  const selectedId = useCRMStore((s) => s.selectedId)
   370→  const jobFilter = useCRMStore((s) => s.jobFilter)
   371→  const setJobFilter = useCRMStore((s) => s.setJobFilter)
   372→  const queryClient = useQueryClient()
   373→
   374→  const [addDialogOpen, setAddDialogOpen] = useState(false)
   375→  const [editJob, setEditJob] = useState<Job | null>(null)
   376→  const [viewMode, setViewMode] = useState<'table' | 'kanban'>('table')
   377→
   378→  // Fetch jobs
   379→  const { data, isLoading, error } = useQuery<{ data: Job[] }>({
   380→    queryKey: ['jobs', jobFilter.search, jobFilter.status, jobFilter.priority],
   381→    queryFn: async () => {
   382→      const params = new URLSearchParams()
   383→      if (jobFilter.search) params.set('search', jobFilter.search)
   384→      if (jobFilter.status && jobFilter.status !== 'All') params.set('status', jobFilter.status)
   385→      if (jobFilter.priority && jobFilter.priority !== 'All') params.set('priority', jobFilter.priority)
   386→      const res = await fetch(`/api/jobs?${params.toString()}`)
   387→      if (!res.ok) throw new Error('Failed to fetch jobs')
   388→      return res.json()
   389→    },
   390→  })
   391→
   392→  const jobs = data?.data || []
   393→
   394→  // Group jobs by status for Kanban
   395→  const kanbanData = useMemo(() => {
   396→    const groups: Record<string, Job[]> = {}
   397→    KANBAN_STATUSES.forEach((s) => {
   398→      groups[s] = []
   399→    })
   400→    jobs.forEach((j) => {
   401→      if (groups[j.status]) {
   402→        groups[j.status].push(j)
   403→      }
   404→    })
   405→    return groups
   406→  }, [jobs])
   407→
   408→  // If a job is selected, show detail view
   409→  if (selectedId) {
   410→    return <JobDetail jobId={selectedId} />
   411→  }
   412→
   413→  function handleViewJob(id: string) {
   414→    navigate('job-detail', id)
   415→  }
   416→
   417→  function handleEditJob(job: Job) {
   418→    setEditJob(job)
   419→    setAddDialogOpen(true)
   420→  }
   421→
   422→  function handleCloseDialog(open: boolean) {
   423→    setAddDialogOpen(open)
   424→    if (!open) setEditJob(null)
   425→  }
   426→
   427→  return (
   428→    <div className="flex flex-1 flex-col gap-6 p-4 md:p-6">
   429→      {/* Header */}
   430→      <div className="space-y-3">
   431→        <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   432→          <div className="flex items-center gap-3">
   433→            <div className="flex h-10 w-10 items-center justify-center rounded-xl bg-amber-100 dark:bg-amber-950">
   434→              <Briefcase className="h-5 w-5 text-amber-600 dark:text-amber-400" />
   435→            </div>
   436→            <div>
   437→              <h1 className="text-2xl font-bold tracking-tight md:text-3xl">Jobs</h1>
   438→              <p className="mt-1 text-sm text-muted-foreground">
   439→                Manage job openings and track recruitment pipeline
   440→                {data && (
   441→                  <span className="ml-1 font-medium text-foreground">
   442→                    ({jobs.length} total)
   443→                  </span>
   444→                )}
   445→              </p>
   446→            </div>
   447→          </div>
   448→          <Button onClick={() => setAddDialogOpen(true)} className="bg-gradient-to-r from-primary to-primary/80 text-primary-foreground shadow-sm hover:from-primary/90 hover:to-primary/70">
   449→            <Plus className="mr-2 h-4 w-4" />
   450→            Add Job
   451→          </Button>
   452→        </div>
   453→        <div className="h-1 w-16 rounded-full bg-gradient-to-r from-amber-400 to-orange-400" />
   454→      </div>
   455→
   456→      {/* Filters */}
   457→      <div className="rounded-lg bg-muted/50 p-2">
   458→        <div className="flex flex-col gap-3 sm:flex-row sm:items-center">
   459→          <div className="relative flex-1 sm:max-w-xs">
   460→            <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   461→            <Input

   463→              className="pl-9"

   466→            />
   467→          </div>

   537→                  <Briefcase className="h-6 w-6 text-amber-400" />
   538→                </div>
   539→                <div className="text-center">
   540→                  <p className="text-sm font-medium text-muted-foreground">No jobs found</p>
   541→                  <p className="mt-1 text-xs text-muted-foreground/70">
   542→                    {jobFilter.search || jobFilter.status !== 'All' || jobFilter.priority !== 'All'
   543→                      ? 'Try adjusting your search or filter criteria'

   545→                  </p>
