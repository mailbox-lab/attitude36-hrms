   152→   152→// ===== Match Score Circle =====
   153→   153→
   154→   154→function MatchScoreCircle({ score }: { score: number }) {
   155→   155→  const radius = 28
   156→   156→  const circumference = 2 * Math.PI * radius
   157→   157→  const offset = circumference - (score / 100) * circumference
   158→   158→  const color = score >= 80 ? 'text-emerald-500' : score >= 65 ? 'text-amber-500' : 'text-rose-500'
   159→   159→  const strokeColor = score >= 80 ? 'stroke-emerald-500' : score >= 65 ? 'stroke-amber-500' : 'stroke-rose-500'
   160→   160→
   161→   161→  return (
   162→   162→    <div className="relative flex h-[72px] w-[72px] items-center justify-center">
   163→   163→      <svg className="h-full w-full -rotate-90" viewBox="0 0 72 72">
   164→   164→        <circle cx="36" cy="36" r={radius} fill="none" strokeWidth="4" className="stroke-muted/30" />
   165→   165→        <motion.circle
   166→   166→          cx="36" cy="36" r={radius} fill="none" strokeWidth="4" strokeLinecap="round"
   167→   167→          className={strokeColor}
   168→   168→          strokeDasharray={circumference}
   169→   169→          initial={{ strokeDashoffset: circumference }}
   170→   170→          animate={{ strokeDashoffset: offset }}
   171→   171→          transition={{ duration: 1.2, ease: 'easeOut', delay: 0.3 }}
   172→   172→        />
   173→   173→      </svg>
   174→   174→      <div className="absolute flex flex-col items-center">
   175→   175→        <span className={`text-base font-bold leading-none ${color}`}>{score}%</span>
   176→   176→        <span className="text-[8px] text-muted-foreground">Match</span>
   177→   177→      </div>
   178→   178→    </div>
   179→   179→  )
   180→   180→}

   204→   204→function StatusTimeline({ currentStatus }: { currentStatus: string }) {
   205→   205→  const currentIndex = STATUS_PIPELINE.indexOf(currentStatus)
   206→   206→  const isTerminal = currentStatus === 'Hired' || currentStatus === 'Rejected'
   207→   207→
   208→   208→  return (
   209→   209→    <div className="flex items-center gap-0 overflow-x-auto pb-2">
   210→   210→      {STATUS_PIPELINE.map((status, idx) => {
   211→   211→        const isCompleted = !isTerminal && idx < currentIndex
   212→   212→        const isCurrent = status === currentStatus
   213→   213→        const isFuture = idx > currentIndex && !isTerminal
   214→   214→        const isTerminalDone = isTerminal && (status === currentStatus || idx < STATUS_PIPELINE.indexOf(currentStatus))
   215→   215→        const dotColor = STATUS_DOT_COLORS[status]
   216→   216→
   217→   217→        return (
   218→   218→          <div key={status} className="group flex items-center">
   219→   219→            <div className="flex flex-col items-center gap-1.5">
   220→   220→              <div
   221→   221→                className={`flex h-8 w-8 shrink-0 items-center justify-center rounded-full text-[11px] font-bold transition-all group-hover:scale-110 ${
   222→   222→                  isCurrent || isTerminalDone
   223→   223→                    ? `${dotColor} text-white ring-4 ring-white shadow-md dark:ring-background`
   224→   224→                    : isFuture
   225→   225→                      ? 'border-2 border-dashed border-muted-foreground/25 text-muted-foreground/40'
   226→   226→                      : `${dotColor} text-white ring-2 ring-white/60 dark:ring-background/60`
   227→   227→                }`}
   228→   228→              >
   229→   229→                {isCompleted || isTerminalDone ? '✓' : idx + 1}
   230→   230→              </div>
   231→   231→              <span
   232→   232→                className={`whitespace-nowrap text-[10px] font-medium transition-colors ${
   233→   233→                  isCurrent ? 'text-foreground' : 'text-muted-foreground'
   234→   234→                }`}
   235→   235→              >
   236→   236→                {status}
   237→   237→              </span>
   238→   238→            </div>
   239→   239→            {idx < STATUS_PIPELINE.length - 1 && (
   240→   240→              <div
   241→   241→                className={`mx-0.5 h-1 w-5 rounded-full transition-colors sm:w-8 ${
   242→   242→                  (isCompleted && !isTerminal) || isTerminalDone
   243→   243→                    ? dotColor.replace('bg-', 'bg-')
   244→   244→                    : 'bg-muted-foreground/15'
   245→   245→                }`}
   246→   246→              />
   247→   247→            )}
   248→   248→          </div>
   249→   249→        )
   250→   250→      })}
   251→   251→    </div>
   252→   252→  )
   253→   253→}

   285→   285→    queryFn: async () => {

   324→   324→      hash |= 0
   325→   325→    }

   327→   327→  }, [candidateId])

   353→   353→  const initials = `${candidate.firstName.charAt(0)}${candidate.lastName.charAt(0)}`.toUpperCase()
   354→   354→  const skills = candidate.skills
   355→   355→    ? candidate.skills.split(',').map((s) => s.trim()).filter(Boolean)
   356→   356→    : []
   357→   357→  const interviews = interviewsData?.data || []
   358→   358→
   359→   359→  const handleScheduleInterview = () => {
   360→   360→    navigate('interviews')
   361→   361→    toast.info('Select a job to schedule interview')
   362→   362→  }

   396→   396→              {candidate.source && (
