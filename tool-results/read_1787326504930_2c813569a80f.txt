   143→// ===== Helper =====

   145→function getInitials(firstName: string, lastName: string) {
   146→  return `${firstName.charAt(0)}${lastName.charAt(0)}`.toUpperCase()
   147→}

   149→function parseSkills(skills: string): string[] {
   150→  return skills
   151→    ? skills
   152→        .split(',')
   153→        .map((s) => s.trim())
   154→        .filter(Boolean)
   155→        .slice(0, 4)
   156→    : []
   157→}

   161→function KanbanCard({
   162→  candidate,
   163→  onView,
   164→  onEdit,
   165→  isSelected,
   166→  onToggleSelect,
   167→}: {
   168→  candidate: Candidate
   169→  onView: () => void

   171→  isSelected: boolean

   175→  const initials = getInitials(candidate.firstName, candidate.lastName)

   194→      <div className="flex items-start justify-between gap-2">
   195→        <div className="flex items-center gap-2 min-w-0">

   197→            <AvatarFallback className="bg-primary/10 text-primary text-[10px]">
   198→              {initials}
   199→            </AvatarFallback>
   200→          </Avatar>
   201→          <div className="min-w-0">
   202→            <button
   203→              onClick={onView}
   204→              className="truncate text-sm font-medium leading-tight hover:underline"

   206→              {candidate.firstName} {candidate.lastName}

   225→              <Eye className="mr-2 h-3.5 w-3.5" /> View Details

   229→            </DropdownMenuItem>
   230→          </DropdownMenuContent>
   231→        </DropdownMenu>
   232→      </div>
   233→      {skills.length > 0 && (
   234→        <div className="mt-2 flex flex-wrap gap-1">
   235→          {skills.map((skill) => (
   236→            <span
   237→              key={skill}
   238→              className="inline-flex items-center rounded bg-muted px-1.5 py-0.5 text-[10px] font-medium text-muted-foreground"
   239→            >
   240→              {skill}
   241→            </span>

   243→        </div>

   245→      {(candidate.location || candidate.experience != null) && (
   246→        <div className="mt-2 flex items-center gap-3 text-[10px] text-muted-foreground">
   247→          {candidate.experience != null && (
   248→            <span>{candidate.experience} yrs exp</span>
   249→          )}
   250→          {candidate.location && (
   251→            <span className="flex items-center gap-0.5">
   252→              <MapPin className="h-2.5 w-2.5" /> {candidate.location}
   253→            </span>
   254→          )}
   255→        </div>
   256→      )}
   257→    </div>
   258→  )

   350→export function CandidatesPage() {
   351→  const navigate = useCRMStore((s) => s.navigate)
   352→  const selectedId = useCRMStore((s) => s.selectedId)
   353→  const candidateFilter = useCRMStore((s) => s.candidateFilter)
   354→  const setCandidateFilter = useCRMStore((s) => s.setCandidateFilter)
   355→  const queryClient = useQueryClient()
   356→
   357→  // Date range helpers
   358→  const todayStr = new Date().toISOString().split('T')[0]
   359→  const getTodayRange = useCallback(() => ({ fromDate: todayStr, toDate: todayStr }), [todayStr])
   360→  const getThisWeekRange = useCallback(() => {
   361→    const now = new Date()
   362→    const dayOfWeek = now.getDay()
   363→    const monday = new Date(now)
   364→    monday.setDate(now.getDate() - ((dayOfWeek + 6) % 7))
   365→    const sunday = new Date(monday)
   366→    sunday.setDate(monday.getDate() + 6)
   367→    return { fromDate: monday.toISOString().split('T')[0], toDate: sunday.toISOString().split('T')[0] }
   368→  }, [])
   369→  const getThisMonthRange = useCallback(() => {
   370→    const now = new Date()
   371→    const firstDay = new Date(now.getFullYear(), now.getMonth(), 1)
   372→    const lastDay = new Date(now.getFullYear(), now.getMonth() + 1, 0)
   373→    return { fromDate: firstDay.toISOString().split('T')[0], toDate: lastDay.toISOString().split('T')[0] }
   374→  }, [])
   375→  const hasDateFilter = !!(candidateFilter.fromDate && candidateFilter.toDate)
   376→
   377→  const [addDialogOpen, setAddDialogOpen] = useState(false)
   378→  const [importOpen, setImportOpen] = useState(false)
   379→  const [editCandidate, setEditCandidate] = useState<Candidate | null>(null)
   380→  const [activeTab, setActiveTab] = useState('list')
   381→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set())
   382→  const [compareIds, setCompareIds] = useState<string[] | null>(null)
   383→
   384→  const selectedIdsArray = useMemo(() => Array.from(selectedIds), [selectedIds])
   385→
   386→  const toggleSelect = useCallback((id: string) => {
   387→    setSelectedIds((prev) => {
   388→      const next = new Set(prev)
   389→      if (next.has(id)) {
   390→        next.delete(id)
   391→      } else {
   392→        next.add(id)
   393→      }
   394→      return next
   395→    })
   396→  }, [])
   397→
   398→  const deselectAll = useCallback(() => {
   399→    setSelectedIds(new Set())
   400→  }, [])
   401→
   402→  function handleBulkSuccess() {
   403→    setSelectedIds(new Set())
   404→    queryClient.invalidateQueries({ queryKey: ['candidates'] })
   405→  }
   406→
   407→  // Fetch candidates with filters
   408→  const { data, isLoading, error } = useQuery<{
   409→    data: Candidate[]
   410→    pagination: { page: number; limit: number; total: number; totalPages: number }
   411→  }>({
   412→    queryKey: ['candidates', candidateFilter],
   413→    queryFn: async () => {
   414→      const params = new URLSearchParams()
   415→      if (candidateFilter.search) params.set('search', candidateFilter.search)
   416→      if (candidateFilter.status && candidateFilter.status !== 'All')
   417→        params.set('status', candidateFilter.status)
   418→      if (candidateFilter.source && candidateFilter.source !== 'All')
   419→        params.set('source', candidateFilter.source)
   420→      if (candidateFilter.fromDate) params.set('fromDate', candidateFilter.fromDate)
   421→      if (candidateFilter.toDate) params.set('toDate', candidateFilter.toDate)
   422→      const res = await fetch(`/api/candidates?${params.toString()}`)
   423→      if (!res.ok) throw new Error('Failed to fetch candidates')
   424→      return res.json()
   425→    },
   426→  })
   427→
   428→  const candidates = data?.data || []
   429→
   430→  const selectAll = useCallback(() => {
   431→    setSelectedIds(new Set(candidates.map((c) => c.id)))
   432→  }, [candidates])
   433→
   434→  const isAllSelected = candidates.length > 0 && candidates.every((c) => selectedIds.has(c.id))
   435→  const isSomeSelected = !isAllSelected && candidates.some((c) => selectedIds.has(c.id))

   441→        method: 'PUT',
   442→        headers: { 'Content-Type': 'application/json' },
   443→        body: JSON.stringify({ status }),
   444→      })
   445→      if (!res.ok) throw new Error('Failed to update status')
   446→      return res.json()
   447→    },
   448→    onSuccess: () => {
   449→      queryClient.invalidateQueries({ queryKey: ['candidates'] })
   450→      toast.success('Candidate status updated')
   451→    },
   452→    onError: () => {
   453→      toast.error('Failed to update candidate status')
   454→    },
   455→  })

   457→  // Delete mutation

   459→    mutationFn: async (id: string) => {
   460→      const res = await fetch(`/api/candidates/${id}`, { method: 'DELETE' })

   462→      return res.json()

   464→    onSuccess: () => {
   465→      queryClient.invalidateQueries({ queryKey: ['candidates'] })

   511→
   512→  // If a candidate is selected, show detail view
   513→  if (selectedId) {

   515→  }
   516→
   517→  return (
   518→    <div className="flex flex-1 flex-col gap-6 p-4 md:p-6">
   519→      {/* Header */}
   520→      <div className="space-y-3">
   521→        <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   522→          <div className="flex items-center gap-3">
   523→            <div className="flex h-10 w-10 items-center justify-center rounded-xl bg-emerald-100 dark:bg-emerald-950">
   524→              <Users className="h-5 w-5 text-emerald-600 dark:text-emerald-400" />
   525→            </div>
   526→            <div>
   527→              <h1 className="text-2xl font-bold tracking-tight md:text-3xl">Candidates</h1>
   528→              <p className="mt-1 text-sm text-muted-foreground">
   529→                Manage and track your candidate pipeline

   531→                  <span className="ml-1 font-medium text-foreground">

   533→                  </span>

   539→          <Button
   540→            variant="outline"
   541→            size="sm"
   542→            className="gap-1.5 border-emerald-300 text-emerald-700 hover:bg-emerald-50 hover:text-emerald-800 dark:border-emerald-700 dark:text-emerald-400 dark:hover:bg-emerald-950/50"

   571→            <Download className="h-4 w-4" />
