   371→        next.delete(id)

   373→        next.add(id)

   390→    data: Candidate[]

   393→    queryKey: ['candidates', candidateFilter],

   398→        params.set('status', candidateFilter.status)

   400→        params.set('source', candidateFilter.source)
   401→      const res = await fetch(`/api/candidates?${params.toString()}`)

   403→      return res.json()
   404→    },
   405→  })

   444→      queryClient.invalidateQueries({ queryKey: ['candidates'] })
