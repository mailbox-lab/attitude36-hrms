   138→function formatCurrency(amount: number | null | undefined) {
   139→  if (amount == null) return '—'
   140→  if (amount >= 100000) {
   141→    return `₹${(amount / 100000).toFixed(2)} LPA`
   142→  }
   143→  return `₹${amount.toLocaleString('en-IN')}`
   144→}
   145→
   146→function formatDate(dateStr: string | null) {
   147→  if (!dateStr) return '—'
   148→  const date = new Date(dateStr)
   149→  return date.toLocaleDateString('en-IN', {
   150→    day: '2-digit',
   151→    month: 'short',
   152→    year: 'numeric',
   153→  })
   154→}
