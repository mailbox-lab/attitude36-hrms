   116→// ===== Helper =====
   117→
   118→function getInitials(firstName: string, lastName: string) {
   119→  return `${firstName.charAt(0)}${lastName.charAt(0)}`.toUpperCase()
   120→}
   121→
   122→function parseSkills(skills: string): string[] {
   123→  return skills
   124→    ? skills
   125→        .split(',')
   126→        .map((s) => s.trim())
   127→        .filter(Boolean)
   128→        .slice(0, 4)
   129→    : []
   130→}

   132→// ===== Kanban Candidate Card =====
   133→
   134→function KanbanCard({
   135→  candidate,
   136→  onView,
   137→  onEdit,
   138→  isSelected,
   139→  onToggleSelect,
   140→}: {
   141→  candidate: Candidate
   142→  onView: () => void
   143→  onEdit: () => void
   144→  isSelected: boolean
   145→  onToggleSelect: () => void
   146→}) {
   147→  const skills = parseSkills(candidate.skills)
   148→  const initials = getInitials(candidate.firstName, candidate.lastName)
   149→
   150→  return (
   151→    <div
   152→      className={`group relative rounded-lg border border-l-4 bg-card p-3.5 shadow-sm transition-all hover:-translate-y-0.5 hover:shadow-md ${
   153→        STATUS_BORDER_COLORS[candidate.status] || 'border-l-gray-500'
   154→      } ${isSelected ? 'ring-2 ring-primary/50 bg-primary/5' : ''}`}
   155→    >
   156→      <div className={`absolute right-2 top-2 z-10 transition-opacity ${isSelected ? 'opacity-100' : 'opacity-0 group-hover:opacity-100'}`}>
   157→        <Checkbox
   158→          checked={isSelected}
   159→          onCheckedChange={(e) => {
   160→            e.stopPropagation()
   161→            onToggleSelect()
   162→          }}
   163→          aria-label={`Select ${candidate.firstName} ${candidate.lastName}`}
   164→          onClick={(e) => e.stopPropagation()}
   165→        />
   166→      </div>
   167→      <div className="flex items-start justify-between gap-2">
   168→        <div className="flex items-center gap-2 min-w-0">
   169→          <Avatar className="h-7 w-7 shrink-0 text-xs">
   170→            <AvatarFallback className="bg-primary/10 text-primary text-[10px]">
   171→              {initials}
   172→            </AvatarFallback>
   173→          </Avatar>
   174→          <div className="min-w-0">
   175→            <button
   176→              onClick={onView}
   177→              className="truncate text-sm font-medium leading-tight hover:underline"
   178→            >
   179→              {candidate.firstName} {candidate.lastName}
   180→            </button>
   181→            {candidate.title && (
   182→              <p className="truncate text-[11px] text-muted-foreground">{candidate.title}</p>
   183→            )}
   184→          </div>
   185→        </div>
   186→        <DropdownMenu>
   187→          <DropdownMenuTrigger asChild>
   188→            <Button
   189→              variant="ghost"
   190→              size="icon"
   191→              className="h-6 w-6 shrink-0 opacity-0 transition-opacity group-hover:opacity-100"
   192→            >
   193→              <MoreHorizontal className="h-3.5 w-3.5" />
   194→            </Button>
   195→          </DropdownMenuTrigger>
   196→          <DropdownMenuContent align="end">
   197→            <DropdownMenuItem onClick={onView}>
   198→              <Eye className="mr-2 h-3.5 w-3.5" /> View Details
   199→            </DropdownMenuItem>
   200→            <DropdownMenuItem onClick={onEdit}>
   201→              <Pencil className="mr-2 h-3.5 w-3.5" /> Edit
   202→            </DropdownMenuItem>
   203→          </DropdownMenuContent>
   204→        </DropdownMenu>
   205→      </div>
   206→      {skills.length > 0 && (
   207→        <div className="mt-2 flex flex-wrap gap-1">
   208→          {skills.map((skill) => (
   209→            <span
   210→              key={skill}
   211→              className="inline-flex items-center rounded bg-muted px-1.5 py-0.5 text-[10px] font-medium text-muted-foreground"
   212→            >
   213→              {skill}
   214→            </span>
   215→          ))}
   216→        </div>
   217→      )}
   218→      {(candidate.location || candidate.experience != null) && (
   219→        <div className="mt-2 flex items-center gap-3 text-[10px] text-muted-foreground">
   220→          {candidate.experience != null && (
   221→            <span>{candidate.experience} yrs exp</span>
   222→          )}
   223→          {candidate.location && (
   224→            <span className="flex items-center gap-0.5">
   225→              <MapPin className="h-2.5 w-2.5" /> {candidate.location}
   226→            </span>
   227→          )}
   228→        </div>
   229→      )}
   230→    </div>
   231→  )
   232→}

   234→// ===== Kanban Column =====

   236→function KanbanColumn({
   237→  status,
   238→  candidates,
   239→  onViewCandidate,
   240→  onEditCandidate,
   241→  selectedIds,
   242→  onToggleSelect,
   243→}: {
   244→  status: string
   245→  candidates: Candidate[]
   246→  onViewCandidate: (id: string) => void
   247→  onEditCandidate: (candidate: Candidate) => void
   248→  selectedIds: Set<string>
   249→  onToggleSelect: (id: string) => void
   250→}) {
   251→  return (
   252→    <div className="flex w-72 shrink-0 flex-col">
   253→      <div className="mb-3 flex items-center justify-between px-1">
   254→        <div className="flex items-center gap-2">
   255→          <div
   256→            className={`h-2.5 w-2.5 rounded-full ${
   257→              STATUS_COLORS[status]
   258→                ?.replace(/bg-\w+-\d+/, 'bg-\$&')
   259→                ?.split(' ')[0] || 'bg-gray-400'
   260→            }`}
   261→            style={{
   262→              backgroundColor:
   263→                status === 'New'
   264→                  ? '#10b981'
   265→                  : status === 'Screening'
   266→                    ? '#3b82f6'
   267→                    : status === 'Interview'
   268→                      ? '#f59e0b'
   269→                      : status === 'Offer'
   270→                        ? '#8b5cf6'
   271→                        : status === 'Hired'
   272→                          ? '#22c55e'
   273→                          : status === 'Rejected'
   274→                            ? '#ef4444'
   275→                            : '#6b7280',
   276→            }}
   277→          />
   278→          <span className="text-sm font-semibold">{status}</span>
   279→        </div>
   280→        <span className="flex h-5 min-w-5 items-center justify-center rounded-full bg-primary/10 px-1.5 text-[10px] font-bold text-primary">
   281→          {candidates.length}
   282→        </span>
   283→      </div>
   284→      <div className="rounded-t-lg border-t-2 flex max-h-[calc(100vh-320px)] min-h-[120px] flex-col gap-2.5 overflow-y-auto rounded-b-lg bg-muted/30 p-2.5" style={{
   285→        borderTopColor:
   286→          status === 'New'
   287→            ? '#10b981'
   288→            : status === 'Screening'
   289→              ? '#3b82f6'
   290→              : status === 'Interview'
   291→                ? '#f59e0b'
   292→                : status === 'Offer'
   293→                  ? '#8b5cf6'
   294→                  : status === 'Hired'
   295→                    ? '#22c55e'
   296→                    : status === 'Rejected'
   297→                      ? '#ef4444'
   298→                      : '#6b7280',
   299→      }}>
   300→        {candidates.length === 0 ? (
   301→          <div className="flex flex-1 items-center justify-center py-8">
   302→            <p className="text-xs text-muted-foreground/60">No candidates</p>
   303→          </div>
   304→        ) : (
   305→          candidates.map((c) => (
   306→            <KanbanCard
   307→              key={c.id}
   308→              candidate={c}
   309→              onView={() => onViewCandidate(c.id)}
   310→              onEdit={() => onEditCandidate(c)}
   311→              isSelected={selectedIds.has(c.id)}
   312→              onToggleSelect={() => onToggleSelect(c.id)}
   313→            />
   314→          ))
   315→        )}
   316→      </div>
   317→    </div>
   318→  )
   319→}

   323→export function CandidatesPage() {
   324→  const navigate = useCRMStore((s) => s.navigate)
   325→  const selectedId = useCRMStore((s) => s.selectedId)
   326→  const candidateFilter = useCRMStore((s) => s.candidateFilter)
   327→  const setCandidateFilter = useCRMStore((s) => s.setCandidateFilter)
   328→  const queryClient = useQueryClient()
   329→
   330→  const [addDialogOpen, setAddDialogOpen] = useState(false)
   331→  const [editCandidate, setEditCandidate] = useState<Candidate | null>(null)
   332→  const [activeTab, setActiveTab] = useState('list')
   333→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set())
   334→
   335→  const selectedIdsArray = useMemo(() => Array.from(selectedIds), [selectedIds])
   336→
   337→  const toggleSelect = useCallback((id: string) => {
   338→    setSelectedIds((prev) => {
   339→      const next = new Set(prev)
   340→      if (next.has(id)) {
   341→        next.delete(id)
   342→      } else {
   343→        next.add(id)
   344→      }
   345→      return next
   346→    })
   347→  }, [])
   348→
   349→  const deselectAll = useCallback(() => {
   350→    setSelectedIds(new Set())
   351→  }, [])
   352→
   353→  function handleBulkSuccess() {
   354→    setSelectedIds(new Set())
   355→    queryClient.invalidateQueries({ queryKey: ['candidates'] })
   356→  }
   357→
   358→  // Fetch candidates with filters
   359→  const { data, isLoading, error } = useQuery<{
   360→    data: Candidate[]
   361→    pagination: { page: number; limit: number; total: number; totalPages: number }
   362→  }>({
   363→    queryKey: ['candidates', candidateFilter],
   364→    queryFn: async () => {
   365→      const params = new URLSearchParams()
   366→      if (candidateFilter.search) params.set('search', candidateFilter.search)
   367→      if (candidateFilter.status && candidateFilter.status !== 'All')
   368→        params.set('status', candidateFilter.status)
   369→      if (candidateFilter.source && candidateFilter.source !== 'All')
   370→        params.set('source', candidateFilter.source)
   371→      const res = await fetch(`/api/candidates?${params.toString()}`)
   372→      if (!res.ok) throw new Error('Failed to fetch candidates')
   373→      return res.json()
   374→    },
   375→  })
   376→
   377→  const candidates = data?.data || []
   378→
   379→  const selectAll = useCallback(() => {
   380→    setSelectedIds(new Set(candidates.map((c) => c.id)))
   381→  }, [candidates])
   382→
   383→  const isAllSelected = candidates.length > 0 && candidates.every((c) => selectedIds.has(c.id))
   384→  const isSomeSelected = !isAllSelected && candidates.some((c) => selectedIds.has(c.id))
   385→
   386→  // Delete mutation
   387→  const deleteMutation = useMutation({
   388→    mutationFn: async (id: string) => {
   389→      const res = await fetch(`/api/candidates/${id}`, { method: 'DELETE' })
   390→      if (!res.ok) throw new Error('Failed to delete candidate')
   391→      return res.json()
   392→    },
   393→    onSuccess: () => {
   394→      queryClient.invalidateQueries({ queryKey: ['candidates'] })
   395→      toast.success('Candidate deleted successfully')
   396→    },
   397→    onError: () => {
   398→      toast.error('Failed to delete candidate')
   399→    },
   400→  })
   401→
   402→  // Group candidates by status for Kanban
   403→  const kanbanData = useMemo(() => {
   404→    const groups: Record<string, Candidate[]> = {}
   405→    ALL_STATUSES.forEach((s) => {
   406→      groups[s] = []
   407→    })
   408→    candidates.forEach((c) => {
   409→      if (groups[c.status]) {
   410→        groups[c.status].push(c)
   411→      } else {
   412→        groups['New'].push(c)
   413→      }
   414→    })
   415→    return groups
   416→  }, [candidates])
   417→
   418→  function handleViewCandidate(id: string) {
   419→    navigate('candidate-detail', id)
   420→  }
   421→
   422→  function handleEditCandidate(candidate: Candidate) {
   423→    setEditCandidate(candidate)
   424→    setAddDialogOpen(true)
   425→  }
   426→
   427→  function handleCloseDialog(open: boolean) {
   428→    setAddDialogOpen(open)
   429→    if (!open) setEditCandidate(null)
   430→  }
   431→
   432→  // If a candidate is selected, show detail view
   433→  if (selectedId) {
   434→    return <CandidateDetail candidateId={selectedId} />
   435→  }
   436→
   437→  return (
   438→    <div className="flex flex-1 flex-col gap-6 p-4 md:p-6">
   439→      {/* Header */}
   440→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   441→        <div>
   442→          <h1 className="text-2xl font-bold tracking-tight md:text-3xl">Candidates</h1>
   443→          <p className="mt-1 text-sm text-muted-foreground">
   444→            Manage and track your candidate pipeline
   445→            {data && (
   446→              <span className="ml-1 font-medium text-foreground">
   447→                ({data.pagination.total} total)
   448→              </span>
   449→            )}
   450→          </p>
   451→        </div>
   452→        <div className="flex gap-2">
   453→          <Button
   454→            variant="outline"
   455→            size="sm"
   456→            className="gap-1.5"
   457→            onClick={() => {
   458→              const csvData = candidates.map((c) => ({
   459→                Name: `${c.firstName} ${c.lastName}`,
   460→                Email: c.email || '',
   461→                Phone: c.phone || '',
   462→                Title: c.title || '',
   463→                Status: c.status,
   464→                Source: c.source || '',
   465→                Experience: c.experience ?? '',
   466→                Location: c.location || '',
   467→                Skills: c.skills,
   468→                'Current CTC': c.currentCTC ?? '',
   469→                'Expected CTC': c.expectedCTC ?? '',
   470→                'Notice Period': c.noticePeriod ?? '',
   471→                'Job Title': c.job?.title || '',
   472→              }))
   473→              exportToCSV(csvData, 'candidates')
   474→            }}
   475→          >
   476→            <Download className="h-4 w-4" />
   477→            Export CSV
   478→          </Button>
   479→          <Button onClick={() => setAddDialogOpen(true)} className="bg-gradient-to-r from-primary to-primary/80 shadow-md hover:shadow-lg">
   480→            <Plus className="mr-2 h-4 w-4" />
   481→            {selectedIds.size > 0 ? `Add Candidate (${selectedIds.size} selected)` : 'Add Candidate'}
   482→          </Button>
   483→          {candidates.length > 0 && (
   484→            <div className="flex items-center gap-2">
   485→              <Checkbox
   486→                checked={isAllSelected}
   487→                ref={(el) => {
   488→                  if (el) {
   489→                    (el as unknown as { indeterminate: boolean }).indeterminate = isSomeSelected
   490→                  }
   491→                }}
   492→                onCheckedChange={(checked) => {
   493→                  if (checked) selectAll()
   494→                  else deselectAll()
   495→                }}
   496→                aria-label="Select all candidates"
   497→              />
   498→              <span className="hidden text-xs text-muted-foreground sm:inline">Select All</span>
   499→            </div>
   500→          )}
   501→        </div>
   502→      </div>
   503→
   504→      {/* Filters */}
   505→      <div className="flex flex-col gap-3 rounded-lg bg-muted/50 p-2 sm:flex-row sm:items-center">
   506→        <div className="relative flex-1 sm:max-w-xs">
   507→          <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   508→          <Input
   509→            placeholder="Search candidates..."
   510→            className="border-0 bg-background pl-9 shadow-sm"
   511→            value={candidateFilter.search}
   512→            onChange={(e) => setCandidateFilter({ search: e.target.value })}
   513→          />
   514→        </div>
   515→        <Select
   516→          value={candidateFilter.status}
   517→          onValueChange={(val) => setCandidateFilter({ status: val })}
   518→        >
   519→          <SelectTrigger className="w-full border-0 bg-background shadow-sm sm:w-[160px]">
   520→            <SelectValue placeholder="Status" />
   521→          </SelectTrigger>
   522→          <SelectContent>
   523→            <SelectItem value="All">All Statuses</SelectItem>
   524→            {ALL_STATUSES.map((status) => (
   525→              <SelectItem key={status} value={status}>
   526→                {status}
   527→              </SelectItem>
   528→            ))}
   529→          </SelectContent>
   530→        </Select>
   531→        <Select
   532→          value={candidateFilter.source}
   533→          onValueChange={(val) => setCandidateFilter({ source: val })}
   534→        >
   535→          <SelectTrigger className="w-full border-0 bg-background shadow-sm sm:w-[160px]">
   536→            <SelectValue placeholder="Source" />
   537→          </SelectTrigger>
   538→          <SelectContent>
   539→            <SelectItem value="All">All Sources</SelectItem>
   540→            {SOURCE_OPTIONS.map((source) => (
   541→              <SelectItem key={source} value={source}>
   542→                {source}
   543→              </SelectItem>
   544→            ))}
   545→          </SelectContent>
   546→        </Select>
   547→      </div>
   548→
   549→      {/* Bulk Actions Bar */}
   550→      <BulkActionsBar
   551→        selectedIds={selectedIdsArray}
   552→        onClearSelection={deselectAll}
   553→        onSuccess={handleBulkSuccess}
   554→      />
   555→
   556→      {/* Tabs: List / Pipeline */}
   557→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   558→        <TabsList>
   559→          <TabsTrigger value="list" className="gap-1.5">
   560→            <LayoutList className="h-4 w-4" />
   561→            List
   562→          </TabsTrigger>
   563→          <TabsTrigger value="pipeline" className="gap-1.5">
   564→            <Kanban className="h-4 w-4" />
   565→            Pipeline
   566→          </TabsTrigger>
   567→        </TabsList>
   568→
   569→        {/* List View */}
   570→        <TabsContent value="list" className="mt-4">
   571→          {isLoading ? (
   572→            <Card>
   573→              <CardContent className="p-6">
   574→                <div className="flex items-center justify-center gap-3">
   575→                  <div className="h-5 w-5 animate-spin rounded-full border-2 border-primary border-t-transparent" />
   576→                  <p className="text-sm text-muted-foreground">Loading candidates...</p>
   577→                </div>
   578→              </CardContent>
   579→            </Card>
   580→          ) : error ? (
   581→            <Card>
   582→              <CardContent className="p-6">
   583→                <p className="text-center text-sm text-destructive">
   584→                  Failed to load candidates. Please try again.
   585→                </p>
   586→              </CardContent>

   772→      </Tabs>
   773→
   774→      {/* Add/Edit Dialog */}
   775→      <AddCandidateDialog
