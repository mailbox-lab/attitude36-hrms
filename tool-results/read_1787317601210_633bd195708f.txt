   152→// ===== Match Score Circle =====
   153→
   154→function MatchScoreCircle({ score }: { score: number }) {
   155→  const radius = 28
   156→  const circumference = 2 * Math.PI * radius
   157→  const offset = circumference - (score / 100) * circumference
   158→  const color = score >= 80 ? 'text-emerald-500' : score >= 65 ? 'text-amber-500' : 'text-rose-500'
   159→  const strokeColor = score >= 80 ? 'stroke-emerald-500' : score >= 65 ? 'stroke-amber-500' : 'stroke-rose-500'
   160→
   161→  return (
   162→    <div className="relative flex h-[72px] w-[72px] items-center justify-center">
   163→      <svg className="h-full w-full -rotate-90" viewBox="0 0 72 72">
   164→        <circle cx="36" cy="36" r={radius} fill="none" strokeWidth="4" className="stroke-muted/30" />
   165→        <motion.circle
   166→          cx="36" cy="36" r={radius} fill="none" strokeWidth="4" strokeLinecap="round"
   167→          className={strokeColor}
   168→          strokeDasharray={circumference}
   169→          initial={{ strokeDashoffset: circumference }}
   170→          animate={{ strokeDashoffset: offset }}
   171→          transition={{ duration: 1.2, ease: 'easeOut', delay: 0.3 }}
   172→        />
   173→      </svg>
   174→      <div className="absolute flex flex-col items-center">
   175→        <span className={`text-base font-bold leading-none ${color}`}>{score}%</span>
   176→        <span className="text-[8px] text-muted-foreground">Match</span>
   177→      </div>
   178→    </div>
   179→  )
   180→}

   202→// ===== Status Timeline =====
   203→
   204→function StatusTimeline({ currentStatus }: { currentStatus: string }) {
   205→  const currentIndex = STATUS_PIPELINE.indexOf(currentStatus)
   206→  const isTerminal = currentStatus === 'Hired' || currentStatus === 'Rejected'
   207→
   208→  return (
   209→    <div className="flex items-center gap-0 overflow-x-auto pb-2">
   210→      {STATUS_PIPELINE.map((status, idx) => {
   211→        const isCompleted = !isTerminal && idx < currentIndex
   212→        const isCurrent = status === currentStatus
   213→        const isFuture = idx > currentIndex && !isTerminal
   214→        const isTerminalDone = isTerminal && (status === currentStatus || idx < STATUS_PIPELINE.indexOf(currentStatus))
   215→        const dotColor = STATUS_DOT_COLORS[status]
   216→
   217→        return (
   218→          <div key={status} className="group flex items-center">
   219→            <div className="flex flex-col items-center gap-1.5">
   220→              <div
   221→                className={`flex h-8 w-8 shrink-0 items-center justify-center rounded-full text-[11px] font-bold transition-all group-hover:scale-110 ${
   222→                  isCurrent || isTerminalDone
   223→                    ? `${dotColor} text-white ring-4 ring-white shadow-md dark:ring-background`
   224→                    : isFuture
   225→                      ? 'border-2 border-dashed border-muted-foreground/25 text-muted-foreground/40'
   226→                      : `${dotColor} text-white ring-2 ring-white/60 dark:ring-background/60`
   227→                }`}
   228→              >
   229→                {isCompleted || isTerminalDone ? '✓' : idx + 1}
   230→              </div>
   231→              <span
   232→                className={`whitespace-nowrap text-[10px] font-medium transition-colors ${
   233→                  isCurrent ? 'text-foreground' : 'text-muted-foreground'
   234→                }`}
   235→              >
   236→                {status}
   237→              </span>
   238→            </div>
   239→            {idx < STATUS_PIPELINE.length - 1 && (
   240→              <div
   241→                className={`mx-0.5 h-1 w-5 rounded-full transition-colors sm:w-8 ${
   242→                  (isCompleted && !isTerminal) || isTerminalDone
   243→                    ? dotColor.replace('bg-', 'bg-')
   244→                    : 'bg-muted-foreground/15'
   245→                }`}
   246→              />
   247→            )}
   248→          </div>
   249→        )
   250→      })}
   251→    </div>
   252→  )
   253→}

   287→      if (!res.ok) throw new Error('Failed to fetch candidate')
   288→      return res.json()
   289→    },

   307→      return res.json()

   310→      queryClient.invalidateQueries({ queryKey: ['candidates'] })

   321→    let hash = 0
   322→    for (let i = 0; i < candidateId.length; i++) {
   323→      hash = ((hash << 5) - hash) + candidateId.charCodeAt(i)
   324→      hash |= 0
   325→    }
   326→    return 60 + Math.abs(hash) % 36 // 60-95
   327→  }, [candidateId])
   328→
   329→  if (isLoading) {
   330→    return (
   331→      <div className="flex flex-1 items-center justify-center p-8">

   335→        </div>
   336→      </div>
   337→    )
   338→  }
   339→
   340→  if (error || !candidate) {

   351→

   378→        <ArrowLeft className="mr-2 h-4 w-4" />
   379→        Back to Candidates
