   201→   201→function getRelativeTime(dateStr: string): string {
   202→   202→  const now = Date.now()
   203→   203→  const then = new Date(dateStr).getTime()
   204→   204→  const diffMs = now - then
   205→   205→  const diffSeconds = Math.floor(diffMs / 1000)
   206→   206→  const diffMinutes = Math.floor(diffSeconds / 60)
   207→   207→  const diffHours = Math.floor(diffMinutes / 60)
   208→   208→  const diffDays = Math.floor(diffHours / 24)
   209→   209→  const diffWeeks = Math.floor(diffDays / 7)
   210→   210→
   211→   211→  if (diffSeconds < 60) return 'Just now'
   212→   212→  if (diffMinutes === 1) return '1 min ago'
   213→   213→  if (diffMinutes < 60) return `${diffMinutes} min ago`
   214→   214→  if (diffHours === 1) return '1 hour ago'
   215→   215→  if (diffHours < 24) return `${diffHours} hours ago`
   216→   216→  if (diffDays === 1) return 'Yesterday'
   217→   217→  if (diffDays < 7) return `${diffDays} days ago`
   218→   218→  if (diffWeeks === 1) return '1 week ago'
   219→   219→  return `${diffWeeks} weeks ago`
   220→   220→}

   222→   222→function isToday(dateStr: string): boolean {
   223→   223→  const d = new Date(dateStr)
   224→   224→  const now = new Date()
   225→   225→  return d.getDate() === now.getDate() &&
   226→   226→    d.getMonth() === now.getMonth() &&
   227→   227→    d.getFullYear() === now.getFullYear()
   228→   228→}
   229→   229→
   230→   230→function isThisWeek(dateStr: string): boolean {
   231→   231→  const d = new Date(dateStr)
   232→   232→  const now = new Date()
   233→   233→  const startOfWeek = new Date(now)
   234→   234→  startOfWeek.setDate(now.getDate() - now.getDay())
   235→   235→  startOfWeek.setHours(0, 0, 0, 0)
   236→   236→  return d >= startOfWeek
   237→   237→}
   238→   238→
   239→   239→function isThisMonth(dateStr: string): boolean {
   240→   240→  const d = new Date(dateStr)
   241→   241→  const now = new Date()
   242→   242→  return d.getMonth() === now.getMonth() && d.getFullYear() === now.getFullYear()
   243→   243→}
   244→   244→
   245→   245→function getDateRangeFilter(dateRange: string, customFrom?: string, customTo?: string): { from?: string; to?: string } {
   246→   246→  if (dateRange === 'today') {
   247→   247→    const today = new Date().toISOString().split('T')[0]
   248→   248→    return { from: today, to: today }
   249→   249→  }
   250→   250→  if (dateRange === 'week') {
   251→   251→    const now = new Date()
   252→   252→    const start = new Date(now)
   253→   253→    start.setDate(now.getDate() - now.getDay())
   254→   254→    start.setHours(0, 0, 0, 0)
   255→   255→    return { from: start.toISOString().split('T')[0], to: now.toISOString().split('T')[0] }
   256→   256→  }
   257→   257→  if (dateRange === 'month' && !customFrom && !customTo) {
   258→   258→    const now = new Date()
   259→   259→    const start = new Date(now.getFullYear(), now.getMonth(), 1)
   260→   260→    return { from: start.toISOString().split('T')[0], to: now.toISOString().split('T')[0] }
   261→   261→  }
   262→   262→  if (dateRange === 'custom' && customFrom && customTo) {
   263→   263→    return { from: customFrom, to: customTo }
   264→   264→  }
   265→   265→  return {}
   266→   266→}
