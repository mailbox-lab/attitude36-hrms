   267→function ChangeIndicator({ value }: { value: number }) {
   268→  if (value === 0) return null
   269→  const isPositive = value > 0
   270→  return (
   271→    <span
   272→      className={`inline-flex items-center gap-0.5 text-xs font-medium ${
   273→        isPositive ? 'text-emerald-600 dark:text-emerald-400' : 'text-red-500 dark:text-red-400'
   274→      }`}
   275→    >
   276→      {isPositive ? (
   277→        <TrendingUp className="h-3 w-3" />
   278→      ) : (
   279→        <TrendingDown className="h-3 w-3" />
   280→      )}
   281→      {Math.abs(value).toFixed(1)}%

   499→function formatRelativeTime(dateStr: string): string {
   500→  const now = new Date()
   501→  const date = new Date(dateStr)
   502→  const diffMs = now.getTime() - date.getTime()
   503→  const diffMins = Math.floor(diffMs / 60000)
   504→  const diffHours = Math.floor(diffMs / 3600000)
   505→  const diffDays = Math.floor(diffMs / 86400000)
   506→
   507→  if (diffMins < 1) return 'Just now'
   508→  if (diffMins < 60) return `${diffMins}m ago`
   509→  if (diffHours < 24) return `${diffHours}h ago`
   510→  if (diffDays < 7) return `${diffDays}d ago`
   511→  return date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short' })
   512→}
