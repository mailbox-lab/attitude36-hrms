   330→   330→function formatInterviewTimeIndicator(dateStr: string, timeStr: string): { text: string; colorClass: string } {
   331→   331→  const now = new Date()
   332→   332→  const [hours, minutes] = timeStr.split(':').map(Number)
   333→   333→  const interviewDate = new Date(dateStr)
   334→   334→  interviewDate.setHours(hours, minutes, 0, 0)
   335→   335→  const diffMs = interviewDate.getTime() - now.getTime()
   336→   336→  const diffHours = diffMs / 3600000
   337→   337→  if (diffMs < 0) return { text: 'Started', colorClass: 'text-red-500' }
   338→   338→  if (diffHours < 2) return { text: `Starts in ${Math.ceil(diffHours)}h`, colorClass: 'text-amber-500' }
   339→   339→  const isToday = interviewDate.toDateString() === now.toDateString()
   340→   340→  if (isToday) return { text: `Today at ${timeStr}`, colorClass: 'text-emerald-500' }
   341→   341→  return { text: `Starts in ${Math.ceil(diffHours)}h`, colorClass: 'text-emerald-500' }
   342→   342→}

   344→   344→function ChangeIndicator({ value }: { value: number }) {
   345→   345→  if (value === 0) return null
   346→   346→  const isPositive = value > 0
   347→   347→  return (
   348→   348→    <span
   349→   349→      className={`inline-flex items-center gap-1 rounded-full px-1.5 py-0.5 text-xs font-medium ${
   350→   350→        isPositive
   351→   351→          ? 'bg-emerald-500/10 text-emerald-600 dark:bg-emerald-500/20 dark:text-emerald-400'
   352→   352→          : 'bg-red-500/10 text-red-500 dark:bg-red-500/20 dark:text-red-400'
   353→   353→      }`}
   354→   354→    >
   355→   355→      {isPositive ? (
   356→   356→        <TrendingUp className="h-3 w-3" />
   357→   357→      ) : (
   358→   358→        <TrendingDown className="h-3 w-3" />
   359→   359→      )}
   360→   360→      {Math.abs(value).toFixed(1)}%
   361→   361→    </span>
   362→   362→  )
   363→   363→}

   686→   686→function formatRelativeTime(dateStr: string): string {
   687→   687→  const now = new Date()
   688→   688→  const date = new Date(dateStr)
   689→   689→  const diffMs = now.getTime() - date.getTime()
   690→   690→  const diffMins = Math.floor(diffMs / 60000)
   691→   691→  const diffHours = Math.floor(diffMs / 3600000)
   692→   692→  const diffDays = Math.floor(diffMs / 86400000)
   693→   693→
   694→   694→  if (diffMins < 1) return 'Just now'
   695→   695→  if (diffMins < 60) return `${diffMins}m ago`
   696→   696→  if (diffHours < 24) return `${diffHours}h ago`
   697→   697→  if (diffDays < 7) return `${diffDays}d ago`
   698→   698→  return date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short' })
   699→   699→}
