   133→// ===== Helper =====
   134→
   135→function getInitials(firstName: string, lastName: string) {
   136→  return `${firstName.charAt(0)}${lastName.charAt(0)}`.toUpperCase()
   137→}
   138→
   139→function parseSkills(skills: string): string[] {
   140→  return skills
   141→    ? skills
   142→        .split(',')
   143→        .map((s) => s.trim())
   144→        .filter(Boolean)
   145→        .slice(0, 4)
   146→    : []
   147→}

   151→function KanbanCard({
   152→  candidate,
   153→  onView,
   154→  onEdit,
   155→  isSelected,
   156→  onToggleSelect,
   157→}: {
   158→  candidate: Candidate
   159→  onView: () => void
   160→  onEdit: () => void
   161→  isSelected: boolean
   162→  onToggleSelect: () => void
   163→}) {
   164→  const skills = parseSkills(candidate.skills)
   165→  const initials = getInitials(candidate.firstName, candidate.lastName)
   166→
   167→  return (
   168→    <div
   169→      className={`group relative rounded-lg border border-l-4 bg-card p-3.5 shadow-sm transition-all hover:-translate-y-0.5 hover:shadow-md ${
   170→        STATUS_BORDER_COLORS[candidate.status] || 'border-l-gray-500'
   171→      } ${isSelected ? 'ring-2 ring-primary/50 bg-primary/5' : ''}`}
   172→    >
   173→      <div className={`absolute right-2 top-2 z-10 transition-opacity ${isSelected ? 'opacity-100' : 'opacity-0 group-hover:opacity-100'}`}>
   174→        <Checkbox
   175→          checked={isSelected}
   176→          onCheckedChange={(e) => {
   177→            e.stopPropagation()
   178→            onToggleSelect()
   179→          }}
   180→          aria-label={`Select ${candidate.firstName} ${candidate.lastName}`}
   181→          onClick={(e) => e.stopPropagation()}
   182→        />
   183→      </div>
   184→      <div className="flex items-start justify-between gap-2">
   185→        <div className="flex items-center gap-2 min-w-0">
   186→          <Avatar className="h-7 w-7 shrink-0 text-xs">
   187→            <AvatarFallback className="bg-primary/10 text-primary text-[10px]">
   188→              {initials}
   189→            </AvatarFallback>
   190→          </Avatar>
   191→          <div className="min-w-0">
   192→            <button
   193→              onClick={onView}
   194→              className="truncate text-sm font-medium leading-tight hover:underline"
   195→            >
   196→              {candidate.firstName} {candidate.lastName}
   197→            </button>
   198→            {candidate.title && (
   199→              <p className="truncate text-[11px] text-muted-foreground">{candidate.title}</p>
   200→            )}
   201→          </div>
   202→        </div>
   203→        <DropdownMenu>
   204→          <DropdownMenuTrigger asChild>
   205→            <Button
   206→              variant="ghost"
   207→              size="icon"
   208→              className="h-6 w-6 shrink-0 opacity-0 transition-opacity group-hover:opacity-100"
   209→            >
   210→              <MoreHorizontal className="h-3.5 w-3.5" />
   211→            </Button>
   212→          </DropdownMenuTrigger>
   213→          <DropdownMenuContent align="end">
   214→            <DropdownMenuItem onClick={onView}>
   215→              <Eye className="mr-2 h-3.5 w-3.5" /> View Details
   216→            </DropdownMenuItem>
   217→            <DropdownMenuItem onClick={onEdit}>
   218→              <Pencil className="mr-2 h-3.5 w-3.5" /> Edit
   219→            </DropdownMenuItem>
   220→          </DropdownMenuContent>
   221→        </DropdownMenu>
   222→      </div>
   223→      {skills.length > 0 && (
   224→        <div className="mt-2 flex flex-wrap gap-1">
   225→          {skills.map((skill) => (
   226→            <span
   227→              key={skill}
   228→              className="inline-flex items-center rounded bg-muted px-1.5 py-0.5 text-[10px] font-medium text-muted-foreground"
   229→            >
   230→              {skill}
   231→            </span>
   232→          ))}
   233→        </div>
   234→      )}
   235→      {(candidate.location || candidate.experience != null) && (
   236→        <div className="mt-2 flex items-center gap-3 text-[10px] text-muted-foreground">
   237→          {candidate.experience != null && (
   238→            <span>{candidate.experience} yrs exp</span>
   239→          )}
   240→          {candidate.location && (
   241→            <span className="flex items-center gap-0.5">
   242→              <MapPin className="h-2.5 w-2.5" /> {candidate.location}
   243→            </span>
   244→          )}
   245→        </div>
   246→      )}
   247→    </div>
   248→  )
   249→}

   340→export function CandidatesPage() {
   341→  const navigate = useCRMStore((s) => s.navigate)
   342→  const selectedId = useCRMStore((s) => s.selectedId)
   343→  const candidateFilter = useCRMStore((s) => s.candidateFilter)
   344→  const setCandidateFilter = useCRMStore((s) => s.setCandidateFilter)
   345→  const queryClient = useQueryClient()
   346→
   347→  const [addDialogOpen, setAddDialogOpen] = useState(false)
   348→  const [editCandidate, setEditCandidate] = useState<Candidate | null>(null)
   349→  const [activeTab, setActiveTab] = useState('list')
   350→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set())
   351→  const [compareIds, setCompareIds] = useState<string[] | null>(null)
   352→
   353→  const selectedIdsArray = useMemo(() => Array.from(selectedIds), [selectedIds])
   354→
   355→  const toggleSelect = useCallback((id: string) => {
   356→    setSelectedIds((prev) => {
   357→      const next = new Set(prev)
   358→      if (next.has(id)) {
   359→        next.delete(id)
   360→      } else {
   361→        next.add(id)
   362→      }
   363→      return next
   364→    })
   365→  }, [])
   366→
   367→  const deselectAll = useCallback(() => {
   368→    setSelectedIds(new Set())
   369→  }, [])
   370→
   371→  function handleBulkSuccess() {
   372→    setSelectedIds(new Set())
   373→    queryClient.invalidateQueries({ queryKey: ['candidates'] })
   374→  }
   375→
   376→  // Fetch candidates with filters
   377→  const { data, isLoading, error } = useQuery<{
   378→    data: Candidate[]
   379→    pagination: { page: number; limit: number; total: number; totalPages: number }
   380→  }>({
   381→    queryKey: ['candidates', candidateFilter],
   382→    queryFn: async () => {
   383→      const params = new URLSearchParams()
   384→      if (candidateFilter.search) params.set('search', candidateFilter.search)
   385→      if (candidateFilter.status && candidateFilter.status !== 'All')
   386→        params.set('status', candidateFilter.status)
   387→      if (candidateFilter.source && candidateFilter.source !== 'All')
   388→        params.set('source', candidateFilter.source)
   389→      const res = await fetch(`/api/candidates?${params.toString()}`)
   390→      if (!res.ok) throw new Error('Failed to fetch candidates')
   391→      return res.json()
   392→    },
   393→  })
   394→
   395→  const candidates = data?.data || []
   396→
   397→  const selectAll = useCallback(() => {
   398→    setSelectedIds(new Set(candidates.map((c) => c.id)))
   399→  }, [candidates])
   400→
   401→  const isAllSelected = candidates.length > 0 && candidates.every((c) => selectedIds.has(c.id))
   402→  const isSomeSelected = !isAllSelected && candidates.some((c) => selectedIds.has(c.id))
   403→
   404→  // Quick status change mutation
   405→  const updateStatusMutation = useMutation({
   406→    mutationFn: async ({ id, status }: { id: string; status: string }) => {
   407→      const res = await fetch(`/api/candidates/${id}`, {
   408→        method: 'PUT',
   409→        headers: { 'Content-Type': 'application/json' },
   410→        body: JSON.stringify({ status }),
   411→      })
   412→      if (!res.ok) throw new Error('Failed to update status')
   413→      return res.json()
   414→    },
   415→    onSuccess: () => {
   416→      queryClient.invalidateQueries({ queryKey: ['candidates'] })
   417→      toast.success('Candidate status updated')
   418→    },
   419→    onError: () => {
   420→      toast.error('Failed to update candidate status')
   421→    },
   422→  })
   423→
   424→  // Delete mutation
   425→  const deleteMutation = useMutation({
   426→    mutationFn: async (id: string) => {
   427→      const res = await fetch(`/api/candidates/${id}`, { method: 'DELETE' })
   428→      if (!res.ok) throw new Error('Failed to delete candidate')
   429→      return res.json()
   430→    },
   431→    onSuccess: () => {
   432→      queryClient.invalidateQueries({ queryKey: ['candidates'] })
   433→      toast.success('Candidate deleted successfully')
   434→    },
   435→    onError: () => {
   436→      toast.error('Failed to delete candidate')
   437→    },
   438→  })
   439→
   440→  // Group candidates by status for Kanban
   441→  const kanbanData = useMemo(() => {
   442→    const groups: Record<string, Candidate[]> = {}
   443→    ALL_STATUSES.forEach((s) => {
   444→      groups[s] = []
   445→    })
   446→    candidates.forEach((c) => {
   447→      if (groups[c.status]) {
   448→        groups[c.status].push(c)
   449→      } else {
   450→        groups['New'].push(c)
   451→      }
   452→    })
   453→    return groups
   454→  }, [candidates])
   455→
   456→  function handleViewCandidate(id: string) {
   457→    navigate('candidate-detail', id)
   458→  }
   459→
   460→  function handleEditCandidate(candidate: Candidate) {
   461→    setEditCandidate(candidate)
   462→    setAddDialogOpen(true)
   463→  }
   464→
   465→  function handleCloseDialog(open: boolean) {
   466→    setAddDialogOpen(open)
   467→    if (!open) setEditCandidate(null)
   468→  }
   469→
   470→  function handleCompare() {
   471→    const ids = Array.from(selectedIds).slice(0, 4)
   472→    setCompareIds(ids)
   473→  }
   474→
   475→  function handleCloseComparison() {
   476→    setCompareIds(null)
   477→  }
   478→
   479→  // If a candidate is selected, show detail view
   480→  if (selectedId) {
   481→    return <CandidateDetail candidateId={selectedId} />
   482→  }
   483→
   484→  return (
   485→    <div className="flex flex-1 flex-col gap-6 p-4 md:p-6">
   486→      {/* Header */}
   487→      <div className="space-y-3">
   488→        <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   489→          <div className="flex items-center gap-3">
   490→            <div className="flex h-10 w-10 items-center justify-center rounded-xl bg-emerald-100 dark:bg-emerald-950">
   491→              <Users className="h-5 w-5 text-emerald-600 dark:text-emerald-400" />
   492→            </div>
   493→            <div>
   494→              <h1 className="text-2xl font-bold tracking-tight md:text-3xl">Candidates</h1>
   495→              <p className="mt-1 text-sm text-muted-foreground">
   496→                Manage and track your candidate pipeline
   497→                {data && (
   498→                  <span className="ml-1 font-medium text-foreground">
   499→                    ({data.pagination.total} total)
   500→                  </span>
   501→                )}
   502→              </p>
   503→            </div>
   504→          </div>
   505→        <div className="flex gap-2">
   506→          <Button
   507→            variant="outline"
   508→            size="sm"
   509→            className="gap-1.5"
   510→            onClick={() => {
   511→              const csvData = candidates.map((c) => ({
   512→                Name: `${c.firstName} ${c.lastName}`,
   513→                Email: c.email || '',
   514→                Phone: c.phone || '',
   515→                Title: c.title || '',
   516→                Status: c.status,
   517→                Source: c.source || '',
   518→                Experience: c.experience ?? '',
   519→                Location: c.location || '',
   520→                Skills: c.skills,
   521→                'Current CTC': c.currentCTC ?? '',
   522→                'Expected CTC': c.expectedCTC ?? '',
   523→                'Notice Period': c.noticePeriod ?? '',
   524→                'Job Title': c.job?.title || '',
   525→              }))
   526→              exportToCSV(csvData, 'candidates')
   527→            }}
   528→          >
   529→            <Download className="h-4 w-4" />
   530→            Export CSV
   531→          </Button>
   532→          <Button onClick={() => setAddDialogOpen(true)} className="bg-gradient-to-r from-primary to-primary/80 shadow-md hover:shadow-lg">
   533→            <Plus className="mr-2 h-4 w-4" />
   534→            {selectedIds.size > 0 ? `Add Candidate (${selectedIds.size} selected)` : 'Add Candidate'}
   535→          </Button>
   536→          {candidates.length > 0 && (
   537→            <div className="flex items-center gap-2">
   538→              <Checkbox
   539→                checked={isAllSelected}
   540→                ref={(el) => {
   541→                  if (el) {
   542→                    (el as unknown as { indeterminate: boolean }).indeterminate = isSomeSelected
   543→                  }
   544→                }}
   545→                onCheckedChange={(checked) => {
   546→                  if (checked) selectAll()
   547→                  else deselectAll()
   548→                }}
   549→                aria-label="Select all candidates"
   550→              />
   551→              <span className="hidden text-xs text-muted-foreground sm:inline">Select All</span>
   552→            </div>
   553→          )}
   554→        </div>
   555→      </div>
   556→      <div className="h-1 w-16 rounded-full bg-gradient-to-r from-emerald-400 to-teal-400" />
   557→      </div>
   558→
   559→      {/* Filters */}
   560→      <div className="flex flex-col gap-3 rounded-lg bg-muted/50 p-2 sm:flex-row sm:items-center">
   561→        <div className="relative flex-1 sm:max-w-xs">
   562→          <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   563→          <Input
   564→            placeholder="Search candidates..."
   565→            className="border-0 bg-background pl-9 shadow-sm"
   566→            value={candidateFilter.search}
   567→            onChange={(e) => setCandidateFilter({ search: e.target.value })}
   568→          />
   569→        </div>
   570→        <Select
   571→          value={candidateFilter.status}
   572→          onValueChange={(val) => setCandidateFilter({ status: val })}
   573→        >
   574→          <SelectTrigger className="w-full border-0 bg-background shadow-sm sm:w-[160px]">
   575→            <SelectValue placeholder="Status" />
   576→          </SelectTrigger>
   577→          <SelectContent>
   578→            <SelectItem value="All">All Statuses</SelectItem>
   579→            {ALL_STATUSES.map((status) => (
   580→              <SelectItem key={status} value={status}>
   581→                {status}
