   176→function formatDate(dateStr: string) {
   177→  const date = new Date(dateStr)
   178→  return date.toLocaleDateString('en-IN', {
   179→    day: '2-digit',
   180→    month: 'short',
   181→    year: 'numeric',
   182→  })
   183→}
   184→
   185→function formatTime(dateStr: string) {
   186→  const date = new Date(dateStr)
   187→  return date.toLocaleTimeString('en-IN', {
   188→    hour: '2-digit',
   189→    minute: '2-digit',
   190→    hour12: true,
   191→  })
   192→}
   193→
   194→function formatDuration(minutes: number) {
   195→  if (minutes < 60) return `${minutes}m`
   196→  const h = Math.floor(minutes / 60)
   197→  const m = minutes % 60
   198→  return m > 0 ? `${h}h ${m}m` : `${h}h`
   199→}
