   138→   138→function formatCurrency(amount: number | null | undefined) {
   139→   139→  if (amount == null) return '—'
   140→   140→  if (amount >= 100000) {
   141→   141→    return `₹${(amount / 100000).toFixed(2)} LPA`
   142→   142→  }
   143→   143→  return `₹${amount.toLocaleString('en-IN')}`
   144→   144→}
   145→   145→
   146→   146→function formatDate(dateStr: string | null) {
   147→   147→  if (!dateStr) return '—'
   148→   148→  const date = new Date(dateStr)
   149→   149→  return date.toLocaleDateString('en-IN', {
   150→   150→    day: '2-digit',
   151→   151→    month: 'short',
   152→   152→    year: 'numeric',
   153→   153→  })
