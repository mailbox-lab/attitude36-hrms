   132→// ===== Helper =====

   134→function getInitials(firstName: string, lastName: string) {
   135→  return `${firstName.charAt(0)}${lastName.charAt(0)}`.toUpperCase()
   136→}
   137→
   138→function parseSkills(skills: string): string[] {
   139→  return skills
   140→    ? skills
   141→        .split(',')
   142→        .map((s) => s.trim())
   143→        .filter(Boolean)
   144→        .slice(0, 4)
   145→    : []
   146→}

   150→function KanbanCard({
   151→  candidate,
   152→  onView,
   153→  onEdit,
   154→  isSelected,
   155→  onToggleSelect,
   156→}: {
   157→  candidate: Candidate
   158→  onView: () => void
   159→  onEdit: () => void
   160→  isSelected: boolean
   161→  onToggleSelect: () => void
   162→}) {
   163→  const skills = parseSkills(candidate.skills)
   164→  const initials = getInitials(candidate.firstName, candidate.lastName)
   165→
   166→  return (
   167→    <div
   168→      className={`group relative rounded-lg border border-l-4 bg-card p-3.5 shadow-sm transition-all hover:-translate-y-0.5 hover:shadow-md ${
   169→        STATUS_BORDER_COLORS[candidate.status] || 'border-l-gray-500'
   170→      } ${isSelected ? 'ring-2 ring-primary/50 bg-primary/5' : ''}`}
   171→    >
   172→      <div className={`absolute right-2 top-2 z-10 transition-opacity ${isSelected ? 'opacity-100' : 'opacity-0 group-hover:opacity-100'}`}>
   173→        <Checkbox
   174→          checked={isSelected}
   175→          onCheckedChange={(e) => {
   176→            e.stopPropagation()
   177→            onToggleSelect()
   178→          }}
   179→          aria-label={`Select ${candidate.firstName} ${candidate.lastName}`}
   180→          onClick={(e) => e.stopPropagation()}
   181→        />
   182→      </div>
   183→      <div className="flex items-start justify-between gap-2">
   184→        <div className="flex items-center gap-2 min-w-0">
   185→          <Avatar className="h-7 w-7 shrink-0 text-xs">
   186→            <AvatarFallback className="bg-primary/10 text-primary text-[10px]">
   187→              {initials}
   188→            </AvatarFallback>
   189→          </Avatar>
   190→          <div className="min-w-0">
   191→            <button
   192→              onClick={onView}
   193→              className="truncate text-sm font-medium leading-tight hover:underline"
   194→            >
   195→              {candidate.firstName} {candidate.lastName}
   196→            </button>
   197→            {candidate.title && (
   198→              <p className="truncate text-[11px] text-muted-foreground">{candidate.title}</p>
   199→            )}
   200→          </div>
   201→        </div>
   202→        <DropdownMenu>
   203→          <DropdownMenuTrigger asChild>
   204→            <Button
   205→              variant="ghost"
   206→              size="icon"
   207→              className="h-6 w-6 shrink-0 opacity-0 transition-opacity group-hover:opacity-100"
   208→            >
   209→              <MoreHorizontal className="h-3.5 w-3.5" />
   210→            </Button>
   211→          </DropdownMenuTrigger>
   212→          <DropdownMenuContent align="end">
   213→            <DropdownMenuItem onClick={onView}>
   214→              <Eye className="mr-2 h-3.5 w-3.5" /> View Details
   215→            </DropdownMenuItem>
   216→            <DropdownMenuItem onClick={onEdit}>
   217→              <Pencil className="mr-2 h-3.5 w-3.5" /> Edit
   218→            </DropdownMenuItem>
   219→          </DropdownMenuContent>
   220→        </DropdownMenu>
   221→      </div>
   222→      {skills.length > 0 && (
   223→        <div className="mt-2 flex flex-wrap gap-1">
   224→          {skills.map((skill) => (
   225→            <span
   226→              key={skill}
   227→              className="inline-flex items-center rounded bg-muted px-1.5 py-0.5 text-[10px] font-medium text-muted-foreground"
   228→            >
   229→              {skill}
   230→            </span>
   231→          ))}
   232→        </div>
   233→      )}
   234→      {(candidate.location || candidate.experience != null) && (
   235→        <div className="mt-2 flex items-center gap-3 text-[10px] text-muted-foreground">
   236→          {candidate.experience != null && (
   237→            <span>{candidate.experience} yrs exp</span>
   238→          )}
   239→          {candidate.location && (
   240→            <span className="flex items-center gap-0.5">
   241→              <MapPin className="h-2.5 w-2.5" /> {candidate.location}
   242→            </span>
   243→          )}
   244→        </div>
   245→      )}
   246→    </div>
   247→  )
   248→}

   339→export function CandidatesPage() {
   340→  const navigate = useCRMStore((s) => s.navigate)
   341→  const selectedId = useCRMStore((s) => s.selectedId)
   342→  const candidateFilter = useCRMStore((s) => s.candidateFilter)
   343→  const setCandidateFilter = useCRMStore((s) => s.setCandidateFilter)
   344→  const queryClient = useQueryClient()
   345→
   346→  const [addDialogOpen, setAddDialogOpen] = useState(false)
   347→  const [editCandidate, setEditCandidate] = useState<Candidate | null>(null)
   348→  const [activeTab, setActiveTab] = useState('list')
   349→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set())
   350→  const [compareIds, setCompareIds] = useState<string[] | null>(null)
   351→
   352→  const selectedIdsArray = useMemo(() => Array.from(selectedIds), [selectedIds])
   353→
   354→  const toggleSelect = useCallback((id: string) => {
   355→    setSelectedIds((prev) => {
   356→      const next = new Set(prev)
   357→      if (next.has(id)) {
   358→        next.delete(id)
   359→      } else {
   360→        next.add(id)
   361→      }
   362→      return next
   363→    })
   364→  }, [])
   365→
   366→  const deselectAll = useCallback(() => {
   367→    setSelectedIds(new Set())
   368→  }, [])
   369→
   370→  function handleBulkSuccess() {
   371→    setSelectedIds(new Set())
   372→    queryClient.invalidateQueries({ queryKey: ['candidates'] })
   373→  }
   374→
   375→  // Fetch candidates with filters
   376→  const { data, isLoading, error } = useQuery<{
   377→    data: Candidate[]
   378→    pagination: { page: number; limit: number; total: number; totalPages: number }
   379→  }>({
   380→    queryKey: ['candidates', candidateFilter],
   381→    queryFn: async () => {
   382→      const params = new URLSearchParams()
   383→      if (candidateFilter.search) params.set('search', candidateFilter.search)
   384→      if (candidateFilter.status && candidateFilter.status !== 'All')
   385→        params.set('status', candidateFilter.status)
   386→      if (candidateFilter.source && candidateFilter.source !== 'All')
   387→        params.set('source', candidateFilter.source)
   388→      const res = await fetch(`/api/candidates?${params.toString()}`)
   389→      if (!res.ok) throw new Error('Failed to fetch candidates')
   390→      return res.json()
   391→    },
   392→  })
   393→
   394→  const candidates = data?.data || []
   395→
   396→  const selectAll = useCallback(() => {
   397→    setSelectedIds(new Set(candidates.map((c) => c.id)))
   398→  }, [candidates])
   399→
   400→  const isAllSelected = candidates.length > 0 && candidates.every((c) => selectedIds.has(c.id))
   401→  const isSomeSelected = !isAllSelected && candidates.some((c) => selectedIds.has(c.id))
   402→
   403→  // Quick status change mutation
   404→  const updateStatusMutation = useMutation({
   405→    mutationFn: async ({ id, status }: { id: string; status: string }) => {
   406→      const res = await fetch(`/api/candidates/${id}`, {
   407→        method: 'PUT',
   408→        headers: { 'Content-Type': 'application/json' },
   409→        body: JSON.stringify({ status }),
   410→      })
   411→      if (!res.ok) throw new Error('Failed to update status')
   412→      return res.json()
   413→    },
   414→    onSuccess: () => {
   415→      queryClient.invalidateQueries({ queryKey: ['candidates'] })
   416→      toast.success('Candidate status updated')
   417→    },
   418→    onError: () => {
   419→      toast.error('Failed to update candidate status')
   420→    },
   421→  })
   422→
   423→  // Delete mutation
   424→  const deleteMutation = useMutation({
   425→    mutationFn: async (id: string) => {
   426→      const res = await fetch(`/api/candidates/${id}`, { method: 'DELETE' })
   427→      if (!res.ok) throw new Error('Failed to delete candidate')
   428→      return res.json()
   429→    },
   430→    onSuccess: () => {
   431→      queryClient.invalidateQueries({ queryKey: ['candidates'] })
   432→      toast.success('Candidate deleted successfully')
   433→    },
   434→    onError: () => {
   435→      toast.error('Failed to delete candidate')
   436→    },
   437→  })
   438→
   439→  // Group candidates by status for Kanban
   440→  const kanbanData = useMemo(() => {
   441→    const groups: Record<string, Candidate[]> = {}
   442→    ALL_STATUSES.forEach((s) => {
   443→      groups[s] = []
   444→    })
   445→    candidates.forEach((c) => {
   446→      if (groups[c.status]) {
   447→        groups[c.status].push(c)
   448→      } else {
   449→        groups['New'].push(c)
   450→      }
   451→    })
   452→    return groups
   453→  }, [candidates])
   454→
   455→  function handleViewCandidate(id: string) {
   456→    navigate('candidate-detail', id)
   457→  }
   458→
   459→  function handleEditCandidate(candidate: Candidate) {
   460→    setEditCandidate(candidate)
   461→    setAddDialogOpen(true)
   462→  }
   463→
   464→  function handleCloseDialog(open: boolean) {
   465→    setAddDialogOpen(open)
   466→    if (!open) setEditCandidate(null)
   467→  }
   468→
   469→  function handleCompare() {
   470→    const ids = Array.from(selectedIds).slice(0, 4)
   471→    setCompareIds(ids)
   472→  }
   473→
   474→  function handleCloseComparison() {
   475→    setCompareIds(null)
   476→  }
   477→
   478→  // If a candidate is selected, show detail view
   479→  if (selectedId) {
   480→    return <CandidateDetail candidateId={selectedId} />
   481→  }
   482→
   483→  return (
   484→    <div className="flex flex-1 flex-col gap-6 p-4 md:p-6">
   485→      {/* Header */}
   486→      <div className="space-y-3">
   487→        <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   488→          <div className="flex items-center gap-3">
   489→            <div className="flex h-10 w-10 items-center justify-center rounded-xl bg-emerald-100 dark:bg-emerald-950">
   490→              <Users className="h-5 w-5 text-emerald-600 dark:text-emerald-400" />
   491→            </div>
   492→            <div>
   493→              <h1 className="text-2xl font-bold tracking-tight md:text-3xl">Candidates</h1>
   494→              <p className="mt-1 text-sm text-muted-foreground">
   495→                Manage and track your candidate pipeline
   496→                {data && (
   497→                  <span className="ml-1 font-medium text-foreground">
   498→                    ({data.pagination.total} total)
   499→                  </span>
   500→                )}
   501→              </p>
   502→            </div>
   503→          </div>
   504→        <div className="flex gap-2">
   505→          <Button
   506→            variant="outline"
   507→            size="sm"
   508→            className="gap-1.5"
   509→            onClick={() => {
   510→              const csvData = candidates.map((c) => ({
   511→                Name: `${c.firstName} ${c.lastName}`,
   512→                Email: c.email || '',
   513→                Phone: c.phone || '',
   514→                Title: c.title || '',
   515→                Status: c.status,
   516→                Source: c.source || '',
   517→                Experience: c.experience ?? '',
   518→                Location: c.location || '',
   519→                Skills: c.skills,
   520→                'Current CTC': c.currentCTC ?? '',
   521→                'Expected CTC': c.expectedCTC ?? '',
   522→                'Notice Period': c.noticePeriod ?? '',
   523→                'Job Title': c.job?.title || '',
   524→              }))
   525→              exportToCSV(csvData, 'candidates')
   526→            }}
   527→          >
   528→            <Download className="h-4 w-4" />
   529→            Export CSV
   530→          </Button>
   531→          <Button onClick={() => setAddDialogOpen(true)} className="bg-gradient-to-r from-primary to-primary/80 shadow-md hover:shadow-lg">
   532→            <Plus className="mr-2 h-4 w-4" />
   533→            {selectedIds.size > 0 ? `Add Candidate (${selectedIds.size} selected)` : 'Add Candidate'}
   534→          </Button>
   535→          {candidates.length > 0 && (
   536→            <div className="flex items-center gap-2">
   537→              <Checkbox
   538→                checked={isAllSelected}
   539→                ref={(el) => {
   540→                  if (el) {
   541→                    (el as unknown as { indeterminate: boolean }).indeterminate = isSomeSelected
   542→                  }
   543→                }}
   544→                onCheckedChange={(checked) => {
   545→                  if (checked) selectAll()
   546→                  else deselectAll()
   547→                }}
   548→                aria-label="Select all candidates"
   549→              />
   550→              <span className="hidden text-xs text-muted-foreground sm:inline">Select All</span>
   551→            </div>
   552→          )}
   553→        </div>
   554→      </div>
   555→      <div className="h-1 w-16 rounded-full bg-gradient-to-r from-emerald-400 to-teal-400" />
   556→      </div>
   557→
   558→      {/* Filters */}
   559→      <div className="flex flex-col gap-3 rounded-lg bg-muted/50 p-2 sm:flex-row sm:items-center">
   560→        <div className="relative flex-1 sm:max-w-xs">
   561→          <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   562→          <Input
   563→            placeholder="Search candidates..."
   564→            className="border-0 bg-background pl-9 shadow-sm"
   565→            value={candidateFilter.search}
   566→            onChange={(e) => setCandidateFilter({ search: e.target.value })}
   567→          />
   568→        </div>
   569→        <Select
   570→          value={candidateFilter.status}
   571→          onValueChange={(val) => setCandidateFilter({ status: val })}
   572→        >
   573→          <SelectTrigger className="w-full border-0 bg-background shadow-sm sm:w-[160px]">
   574→            <SelectValue placeholder="Status" />
   575→          </SelectTrigger>
   576→          <SelectContent>
   577→            <SelectItem value="All">All Statuses</SelectItem>
   578→            {ALL_STATUSES.map((status) => (
   579→              <SelectItem key={status} value={status}>
   580→                {status}
   581→              </SelectItem>
