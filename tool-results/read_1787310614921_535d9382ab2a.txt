   316→function getGreeting(): string {
   317→  const hour = new Date().getHours()
   318→  if (hour < 12) return 'Good Morning'
   319→  if (hour < 17) return 'Good Afternoon'
   320→  return 'Good Evening'

   323→function formatInterviewTimeIndicator(dateStr: string, timeStr: string): { text: string; colorClass: string } {
   324→  const now = new Date()
   325→  const [hours, minutes] = timeStr.split(':').map(Number)
   326→  const interviewDate = new Date(dateStr)
   327→  interviewDate.setHours(hours, minutes, 0, 0)
   328→  const diffMs = interviewDate.getTime() - now.getTime()
   329→  const diffHours = diffMs / 3600000
   330→  if (diffMs < 0) return { text: 'Started', colorClass: 'text-red-500' }
   331→  if (diffHours < 2) return { text: `Starts in ${Math.ceil(diffHours)}h`, colorClass: 'text-amber-500' }
   332→  const isToday = interviewDate.toDateString() === now.toDateString()
   333→  if (isToday) return { text: `Today at ${timeStr}`, colorClass: 'text-emerald-500' }
   334→  return { text: `Starts in ${Math.ceil(diffHours)}h`, colorClass: 'text-emerald-500' }
   335→}

   337→function ChangeIndicator({ value }: { value: number }) {
   338→  if (value === 0) return null
   339→  const isPositive = value > 0
   340→  return (
   341→    <span
   342→      className={`inline-flex items-center gap-1 rounded-full px-1.5 py-0.5 text-xs font-medium ${
   343→        isPositive
   344→          ? 'bg-emerald-500/10 text-emerald-600 dark:bg-emerald-500/20 dark:text-emerald-400'
   345→          : 'bg-red-500/10 text-red-500 dark:bg-red-500/20 dark:text-red-400'
   346→      }`}
   347→    >
   348→      {isPositive ? (
   349→        <TrendingUp className="h-3 w-3" />
   350→      ) : (
   351→        <TrendingDown className="h-3 w-3" />
   352→      )}
   353→      {Math.abs(value).toFixed(1)}%
   354→    </span>
   355→  )
   356→}

   428→function CustomTooltip({
   429→  active,
   430→  payload,
   431→  label,
   432→}: {
   433→  active?: boolean
   434→  payload?: Array<{ value: number; name: string; color: string }>
   435→  label?: string
   436→}) {
   437→  if (!active || !payload || payload.length === 0) return null
   438→  return (
   439→    <div className="rounded-lg border bg-background px-3 py-2 text-sm shadow-md">
   440→      <p className="font-medium">{label}</p>
   441→      {payload.map((entry, idx) => (
   442→        <p key={idx} style={{ color: entry.color }} className="text-sm">
   443→          {entry.name}: {entry.value}
   444→        </p>
   445→      ))}
   446→    </div>
   447→  )
   448→}

   679→function formatRelativeTime(dateStr: string): string {
   680→  const now = new Date()
   681→  const date = new Date(dateStr)
   682→  const diffMs = now.getTime() - date.getTime()
   683→  const diffMins = Math.floor(diffMs / 60000)
   684→  const diffHours = Math.floor(diffMs / 3600000)
   685→  const diffDays = Math.floor(diffMs / 86400000)
   686→
   687→  if (diffMins < 1) return 'Just now'
   688→  if (diffMins < 60) return `${diffMins}m ago`
   689→  if (diffHours < 24) return `${diffHours}h ago`
   690→  if (diffDays < 7) return `${diffDays}d ago`
   691→  return date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short' })
   692→}
