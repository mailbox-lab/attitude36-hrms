   373→    const attendanceRecords: Array<{
   374→      employeeId: string;
   375→      date: Date;
   376→      clockIn: Date;
   377→      clockOut: Date;
   378→      status: string;
   379→      totalHours: number;
   380→      notes?: string;
   381→    }> = [];
   382→
   383→    for (let dayOffset = 30; dayOffset >= 1; dayOffset--) {
   384→      const date = new Date(now);
   385→      date.setDate(date.getDate() - dayOffset);
   386→      const dayOfWeek = date.getDay();
   387→
   388→      // Skip weekends
   389→      if (dayOfWeek === 0 || dayOfWeek === 6) continue;
   390→
   391→      for (const emp of employees) {
   392→        // Randomize some absences and half-days
   393→        const rand = Math.random();
   394→        if (rand < 0.05) {
   395→          // 5% chance of leave
   396→          attendanceRecords.push({
   397→            employeeId: emp.id,
   398→            date: new Date(date.getFullYear(), date.getMonth(), date.getDate()),
   399→            clockIn: new Date(date.getFullYear(), date.getMonth(), date.getDate(), 0, 0, 0),
   400→            clockOut: new Date(date.getFullYear(), date.getMonth(), date.getDate(), 0, 0, 0),
   401→            status: 'Leave',
   402→            totalHours: 0,
   403→            notes: 'Planned leave',
   404→          });
   405→        } else if (rand < 0.1) {
   406→          // 5% chance of work from home
   407→          const clockIn = new Date(date.getFullYear(), date.getMonth(), date.getDate(), 9 + Math.random(), Math.floor(Math.random() * 30), 0);
   408→          const clockOut = new Date(date.getFullYear(), date.getMonth(), date.getDate(), 17 + Math.random(), Math.floor(Math.random() * 60), 0);
   409→          const totalHours = parseFloat(((clockOut.getTime() - clockIn.getTime()) / (1000 * 60 * 60)).toFixed(2));
   410→          attendanceRecords.push({
   411→            employeeId: emp.id,
   412→            date: new Date(date.getFullYear(), date.getMonth(), date.getDate()),
   413→            clockIn,
   414→            clockOut,
   415→            status: 'WFH',
   416→            totalHours,
   417→          });
   418→        } else if (rand < 0.15) {
   419→          // 5% chance of half-day
   420→          const clockIn = new Date(date.getFullYear(), date.getMonth(), date.getDate(), 9, 0, 0);
   421→          const clockOut = new Date(date.getFullYear(), date.getMonth(), date.getDate(), 13, 30, 0);
   422→          attendanceRecords.push({
   423→            employeeId: emp.id,
   424→            date: new Date(date.getFullYear(), date.getMonth(), date.getDate()),
   425→            clockIn,
   426→            clockOut,
   427→            status: 'Half Day',
   428→            totalHours: 4.5,
   429→          });
   430→        } else {
   431→          // Normal day
   432→          const clockInHour = 8 + Math.floor(Math.random() * 2); // 8 or 9
   433→          const clockInMin = Math.floor(Math.random() * 30);
   434→          const clockOutHour = 17 + Math.floor(Math.random() * 2); // 17 or 18
   435→          const clockOutMin = Math.floor(Math.random() * 60);
   436→          const clockIn = new Date(date.getFullYear(), date.getMonth(), date.getDate(), clockInHour, clockInMin, 0);
   437→          const clockOut = new Date(date.getFullYear(), date.getMonth(), date.getDate(), clockOutHour, clockOutMin, 0);
   438→          const totalHours = parseFloat(((clockOut.getTime() - clockIn.getTime()) / (1000 * 60 * 60)).toFixed(2));
   439→          attendanceRecords.push({
   440→            employeeId: emp.id,
   441→            date: new Date(date.getFullYear(), date.getMonth(), date.getDate()),
   442→            clockIn,
   443→            clockOut,
   444→            status: 'Present',
   445→            totalHours,
   446→          });
   447→        }
   448→      }
   449→    }
   450→
   451→    // Insert attendance in batches
   452→    const batchSize = 50;
   453→    for (let i = 0; i < attendanceRecords.length; i += batchSize) {
   454→      await Promise.all(
   455→        attendanceRecords.slice(i, i + batchSize).map((a) =>
   456→          db.attendance.create({ data: a })
   457→        )
   458→      );
   459→    }
   460→
   461→    // ===== LEAVE BALANCES (current year) =====
   462→    const currentYear = now.getFullYear();
   463→    const leaveBalanceData: Array<{
   464→      employeeId: string;
   465→      year: number;
   466→      type: string;
   467→      total: number;
   468→      used: number;
   469→      remaining: number;
   470→    }> = [];
   471→
   472→    for (const emp of employees) {
   473→      leaveBalanceData.push(
   474→        { employeeId: emp.id, year: currentYear, type: 'Casual Leave', total: 12, used: 0, remaining: 12 },
   475→        { employeeId: emp.id, year: currentYear, type: 'Sick Leave', total: 10, used: 0, remaining: 10 },
   476→        { employeeId: emp.id, year: currentYear, type: 'Earned Leave', total: 15, used: 0, remaining: 15 },
   477→        { employeeId: emp.id, year: currentYear, type: 'Compensatory Off', total: 2, used: 0, remaining: 2 },
   478→      );
   479→    }
   480→
   481→    await Promise.all(
   482→      leaveBalanceData.map((lb) => db.leaveBalance.create({ data: lb }))
   483→    );
   484→
   485→    // ===== LEAVE REQUESTS =====
   486→    const leaveRequestData = [
   487→      { employeeId: employees[0].id, type: 'Casual Leave', startDate: new Date(now.getTime() + 5 * 24 * 60 * 60 * 1000), endDate: new Date(now.getTime() + 7 * 24 * 60 * 60 * 1000), totalDays: 3, reason: 'Family function', status: 'Approved', approvedBy: 'admin' },
   488→      { employeeId: employees[1].id, type: 'Sick Leave', startDate: new Date(now.getTime() - 3 * 24 * 60 * 60 * 1000), endDate: new Date(now.getTime() - 3 * 24 * 60 * 60 * 1000), totalDays: 1, reason: 'Not feeling well, doctor appointment', status: 'Approved', approvedBy: 'priya.sharma@recruitcrm.com' },
   489→      { employeeId: employees[2].id, type: 'Casual Leave', startDate: new Date(now.getTime() + 10 * 24 * 60 * 60 * 1000), endDate: new Date(now.getTime() + 12 * 24 * 60 * 60 * 1000), totalDays: 3, reason: 'Travel plans', status: 'Pending', approvedBy: null },
   490→      { employeeId: employees[3].id, type: 'Earned Leave', startDate: new Date(now.getTime() + 15 * 24 * 60 * 60 * 1000), endDate: new Date(now.getTime() + 19 * 24 * 60 * 60 * 1000), totalDays: 5, reason: 'Vacation with family', status: 'Pending', approvedBy: null },
   491→      { employeeId: employees[4].id, type: 'Casual Leave', startDate: new Date(now.getTime() - 10 * 24 * 60 * 60 * 1000), endDate: new Date(now.getTime() - 10 * 24 * 60 * 60 * 1000), totalDays: 1, reason: 'Personal work', status: 'Rejected', approvedBy: 'priya.sharma@recruitcrm.com' },
   492→      { employeeId: employees[5].id, type: 'Earned Leave', startDate: new Date(now.getTime() + 7 * 24 * 60 * 60 * 1000), endDate: new Date(now.getTime() + 14 * 24 * 60 * 60 * 1000), totalDays: 8, reason: 'International travel', status: 'Pending', approvedBy: null },
   493→    ];
   494→
   495→    await Promise.all(
   496→      leaveRequestData.map((lr) => db.leaveRequest.create({ data: lr }))
   497→    );
   498→
   499→    return NextResponse.json({
   500→      message: 'Seed data created successfully',
   501→      counts: {
   502→        employees: employees.length,
   503→        clients: clients.length,
   504→        jobs: jobs.length,
   505→        candidates: candidates.length,
   506→        interviews: interviewData.length,
   507→        placements: placementData.length,
   508→        attendanceRecords: attendanceRecords.length,
   509→        leaveBalances: leaveBalanceData.length,
   510→        leaveRequests: leaveRequestData.length,
   511→      },
   512→    });
   513→  } catch (error) {
   514→    console.error('Error seeding data:', error);
   515→    return NextResponse.json({ error: 'Failed to seed data' }, { status: 500 });
   516→  }
   517→}
