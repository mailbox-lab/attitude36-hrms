   164→function formatDate(dateStr: string) {
   165→  const date = new Date(dateStr)
   166→  return date.toLocaleDateString('en-IN', {
   167→    day: '2-digit',
   168→    month: 'short',
   169→    year: 'numeric',
   170→  })
   171→}
   172→
   173→function formatTime(dateStr: string) {
   174→  const date = new Date(dateStr)
   175→  return date.toLocaleTimeString('en-IN', {
   176→    hour: '2-digit',
   177→    minute: '2-digit',
   178→    hour12: true,
   179→  })
   180→}
   181→
   182→function formatDuration(minutes: number) {
   183→  if (minutes < 60) return `${minutes}m`
   184→  const h = Math.floor(minutes / 60)
   185→  const m = minutes % 60
   186→  return m > 0 ? `${h}h ${m}m` : `${h}h`
   187→}
