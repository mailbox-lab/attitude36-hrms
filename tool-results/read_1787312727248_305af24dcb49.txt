   163→   163→function formatDate(dateStr: string) {
   164→   164→  const date = new Date(dateStr)
   165→   165→  return date.toLocaleDateString('en-IN', {
   166→   166→    day: '2-digit',
   167→   167→    month: 'short',
   168→   168→    year: 'numeric',
   169→   169→  })
   170→   170→}
   171→   171→
   172→   172→function formatTime(dateStr: string) {
   173→   173→  const date = new Date(dateStr)
   174→   174→  return date.toLocaleTimeString('en-IN', {
   175→   175→    hour: '2-digit',
   176→   176→    minute: '2-digit',
   177→   177→    hour12: true,
   178→   178→  })
   179→   179→}
   180→   180→
   181→   181→function formatDuration(minutes: number) {
   182→   182→  if (minutes < 60) return `${minutes}m`
   183→   183→  const h = Math.floor(minutes / 60)
   184→   184→  const m = minutes % 60
   185→   185→  return m > 0 ? `${h}h ${m}m` : `${h}h`
   186→   186→}
