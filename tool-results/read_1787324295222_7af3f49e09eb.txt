   370→// ===================== Data Management Tab =====================
   371→function DataManagementTab() {
   372→  const [stats, setStats] = useState({
   373→    totalCandidates: 0,
   374→    totalClients: 0,
   375→    totalJobs: 0,
   376→    totalEmployees: 0,
   377→  })
   378→  const [statsLoading, setStatsLoading] = useState(true)
   379→  const [exporting, setExporting] = useState(false)
   380→  const [clearing, setClearing] = useState(false)
   381→  const [reseeding, setReseeding] = useState(false)
   382→
   383→  const fetchStats = useCallback(async () => {
   384→    try {
   385→      setStatsLoading(true)
   386→      const [dashRes, clientRes, empRes] = await Promise.all([
   387→        fetch('/api/dashboard'),
   388→        fetch('/api/clients'),
   389→        fetch('/api/employees?activeOnly=false'),
   390→      ])
   391→
   392→      const dashData = dashRes.ok ? await dashRes.json() : {}
   393→      const clientData = clientRes.ok ? await clientRes.json() : { data: [] }
   394→      const empData = empRes.ok ? await empRes.json() : { data: [] }
   395→
   396→      const clientList = clientData.data || clientData || []
   397→      const empList = empData.data || empData || []
   398→
   399→      // Total clients from API (all, not just active)
   400→      const totalClients = Array.isArray(clientList) ? clientList.length : 0
   401→      const totalEmployees = Array.isArray(empList) ? empList.length : 0
   402→
   403→      // Total jobs: we need open + closed. The dashboard gives us openJobs.
   404→      // For total, we fetch from the jobs API
   405→      const jobsRes = await fetch('/api/jobs')
   406→      const jobsData = jobsRes.ok ? await jobsRes.json() : { data: [] }
   407→      const jobList = jobsData.data || jobsData || []
   408→      const totalJobs = Array.isArray(jobList) ? jobList.length : 0
   409→
   410→      setStats({
   411→        totalCandidates: dashData.totalCandidates || 0,
   412→        totalClients,
   413→        totalJobs,
   414→        totalEmployees,
   415→      })
   416→    } catch {
   417→      // silently fail
   418→    } finally {
   419→      setStatsLoading(false)
   420→    }
   421→  }, [])
   422→
   423→  useEffect(() => {
   424→    fetchStats()
   425→  }, [fetchStats])
   426→
   427→  const handleExport = async () => {
   428→    setExporting(true)
   429→    try {
   430→      const res = await fetch('/api/candidates')
   431→      if (!res.ok) throw new Error('Failed to fetch candidates')
   432→      const json = await res.json()
   433→      const candidates = json.data || json || []
   434→
   435→      if (candidates.length === 0) {
   436→        toast.error('No candidates to export')
   437→        setExporting(false)
   438→        return
   439→      }
   440→
   441→      // Build CSV
   442→      const headers = ['Name', 'Email', 'Phone', 'Title', 'Location', 'Experience', 'Status', 'Source', 'Rating']
   443→      const rows = candidates.map((c: Record<string, unknown>) => [
   444→        `${c.firstName || ''} ${c.lastName || ''}`.trim(),
   445→        c.email || '',
   446→        c.phone || '',
   447→        c.title || '',
   448→        c.location || '',
   449→        c.experience ?? '',
   450→        c.status || '',
   451→        c.source || '',
   452→        c.rating ?? '',
   453→      ])
   454→
   455→      const csvContent = [headers, ...rows]
   456→        .map((row) => row.map((cell) => `"${String(cell).replace(/"/g, '""')}"`).join(','))
   457→        .join('\n')
   458→
   459→      const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' })
   460→      const url = URL.createObjectURL(blob)
   461→      const link = document.createElement('a')
   462→      link.href = url
   463→      link.download = `recruitpro-candidates-${new Date().toISOString().split('T')[0]}.csv`
   464→      link.click()
   465→      URL.revokeObjectURL(url)
   466→
   467→      toast.success(`Exported ${candidates.length} candidates to CSV`)
   468→    } catch {
   469→      toast.error('Failed to export data')
   470→    } finally {
   471→      setExporting(false)
   472→    }
   473→  }
   474→
   475→  const handleClearAll = async () => {
   476→    setClearing(true)
   477→    try {
   478→      // Delete all data by model type
   479→      const models = ['placements', 'interviews', 'leave', 'attendance', 'candidates', 'jobs', 'clients', 'employees']
   480→      await Promise.all(
   481→        models.map(async (model) => {
   482→          // Fetch all IDs first, then delete individually
   483→          try {
   484→            const res = await fetch(`/api/${model}`)
   485→            if (!res.ok) return
   486→            const json = await res.json()
   487→            const items = json.data || json || []
   488→            if (Array.isArray(items)) {
   489→              await Promise.all(
   490→                items.map((item: Record<string, unknown>) =>
   491→                  fetch(`/api/${model}/${item.id}`, { method: 'DELETE' }).catch(() => {})
   492→                )
   493→              )
   494→            }
   495→          } catch {
   496→            // continue
   497→          }
   498→        })
   499→      )
   500→      toast.success('All data has been cleared successfully')
   501→      fetchStats()
   502→    } catch {
   503→      toast.error('Failed to clear data')
   504→    } finally {
   505→      setClearing(false)
   506→    }
   507→  }
   508→
   509→  const handleReseed = async () => {
   510→    setReseeding(true)
   511→    try {
   512→      const res = await fetch('/api/seed', { method: 'POST' })
   513→      if (res.ok) {
   514→        toast.success('Sample data has been reseeded successfully')
   515→        fetchStats()
   516→      } else {
   517→        toast.error('Failed to reseed data')
   518→      }
   519→    } catch {
   520→      toast.error('Failed to reseed data')
   521→    } finally {
   522→      setReseeding(false)
   523→    }
   524→  }
   525→
   526→  return (
   527→    <div className="space-y-6">
   528→      {/* Database Info */}
   529→      <Card>
   530→        <CardHeader>
   531→          <CardTitle className="text-lg">Database Overview</CardTitle>
   532→          <CardDescription>Current statistics for your RecruitPro database.</CardDescription>
   533→        </CardHeader>
   534→        <CardContent>
   535→          {statsLoading ? (
   536→            <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   537→              {Array.from({ length: 4 }).map((_, i) => (
   538→                <div key={i} className="h-20 rounded-lg bg-muted animate-pulse" />
   539→              ))}
   540→            </div>
   541→          ) : (
   542→            <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   543→              <div className="rounded-lg border bg-card p-4 text-center">
   544→                <p className="text-2xl font-bold text-primary">{stats.totalCandidates}</p>
   545→                <p className="text-xs text-muted-foreground mt-1">Total Candidates</p>
   546→              </div>
   547→              <div className="rounded-lg border bg-card p-4 text-center">
   548→                <p className="text-2xl font-bold text-emerald-600 dark:text-emerald-400">{stats.totalClients}</p>
   549→                <p className="text-xs text-muted-foreground mt-1">Total Clients</p>
   550→              </div>
   551→              <div className="rounded-lg border bg-card p-4 text-center">
   552→                <p className="text-2xl font-bold text-amber-600 dark:text-amber-400">{stats.totalJobs}</p>
   553→                <p className="text-xs text-muted-foreground mt-1">Total Jobs</p>
   554→              </div>
   555→              <div className="rounded-lg border bg-card p-4 text-center">
   556→                <p className="text-2xl font-bold text-rose-600 dark:text-rose-400">{stats.totalEmployees}</p>
   557→                <p className="text-xs text-muted-foreground mt-1">Total Employees</p>
   558→              </div>
   559→            </div>
   560→          )}
   561→        </CardContent>
   562→      </Card>
   563→
   564→      {/* Actions */}
   565→      <Card>
   566→        <CardHeader>
   567→          <CardTitle className="text-lg">Data Actions</CardTitle>
   568→          <CardDescription>Export, clear, or reseed your application data.</CardDescription>
   569→        </CardHeader>
   570→        <CardContent className="space-y-4">
   571→          <div className="flex flex-col sm:flex-row gap-3">
   572→            <Button
   573→              variant="outline"
   574→              onClick={handleExport}
   575→              disabled={exporting}
   576→              className="flex-1 sm:flex-none"
   577→            >
   578→              {exporting ? (
   579→                <Loader2 className="h-4 w-4 mr-2 animate-spin" />
   580→              ) : (
   581→                <Download className="h-4 w-4 mr-2" />

   583→              Export All Data (CSV)
