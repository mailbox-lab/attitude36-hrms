    90→// ===== Custom Tooltip =====
    91→
    92→function ChartTooltip({
    93→  active,
    94→  payload,
    95→  label,
    96→}: {
    97→  active?: boolean
    98→  payload?: Array<{ value: number; name: string; color: string }>
    99→  label?: string
   100→}) {
   101→  if (!active || !payload || payload.length === 0) return null
   102→  return (
   103→    <div className="rounded-lg border bg-background px-3 py-2 text-sm shadow-md">
   104→      <p className="font-medium">{label}</p>
   105→      {payload.map((entry, idx) => (
   106→        <p key={idx} style={{ color: entry.color }} className="text-sm">
   107→          {entry.name}: {typeof entry.value === 'number' && entry.name?.toLowerCase().includes('revenue')
   108→            ? `\u20b9${(entry.value / 100000).toFixed(1)}L`
   109→            : entry.value.toLocaleString('en-IN')}
   110→        </p>
   111→      ))}
   112→    </div>
   113→  )
   114→}
