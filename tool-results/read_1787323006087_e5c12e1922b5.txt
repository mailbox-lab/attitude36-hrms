   171→function formatCurrency(amount: number | null | undefined) {
   172→  if (amount == null) return '—'
   173→  if (amount >= 100000) {
   174→    return `₹${(amount / 100000).toFixed(2)} LPA`
   175→  }
   176→  return `₹${amount.toLocaleString('en-IN')}`
   177→}
   178→
   179→function formatCompactCurrency(amount: number) {
   180→  if (amount >= 10000000) {
   181→    return `₹${(amount / 10000000).toFixed(2)} Cr`
   182→  }
   183→  if (amount >= 100000) {
   184→    return `₹${(amount / 100000).toFixed(1)} L`
   185→  }
   186→  if (amount >= 1000) {
   187→    return `₹${(amount / 1000).toFixed(1)}K`
   188→  }
   189→  return `₹${amount.toFixed(0)}`
   190→}
   191→
   192→function formatDate(dateStr: string | null) {
   193→  if (!dateStr) return '—'
   194→  const date = new Date(dateStr)
   195→  return date.toLocaleDateString('en-IN', {
   196→    day: '2-digit',
   197→    month: 'short',
   198→    year: 'numeric',
   199→  })
   200→}

   202→function getInitials(firstName: string, lastName: string) {
   203→  return `${(firstName?.[0] || '').toUpperCase()}${(lastName?.[0] || '').toUpperCase()}`
   204→}

   206→function exportToCSV(placements: Placement[]) {
   207→  const headers = ['Candidate', 'Client', 'Job Title', 'Recruiter', 'Offered CTC', 'Joined Date', 'Status', 'Commission']
   208→  const rows = placements.map((p) => [
   209→    `${p.candidate.firstName} ${p.candidate.lastName}`,
   210→    p.client?.name || 'N/A',
   211→    p.job?.title || 'N/A',
   212→    p.recruiter?.name || 'N/A',
   213→    p.offeredCTC ?? '',
   214→    p.joinedDate ? formatDate(p.joinedDate) : '',
   215→    p.status,
   216→    p.commission ?? '',
   217→  ])
   218→  const csv = [headers, ...rows].map((row) => row.map((cell) => `"${cell}"`).join(',')).join('\n')
   219→  const blob = new Blob([csv], { type: 'text/csv' })
   220→  const url = URL.createObjectURL(blob)
   221→  const a = document.createElement('a')
   222→  a.href = url
   223→  a.download = `placements-${new Date().toISOString().split('T')[0]}.csv`
   224→  a.click()
   225→  URL.revokeObjectURL(url)
   226→}

   335→// ===== Status Pipeline =====
   336→
   337→function StatusPipeline({ placements }: { placements: Placement[] }) {
   338→  const pipelineData = useMemo(() => {
   339→    const total = placements.length || 1
   340→    return ['Offered', 'Accepted', 'Joined', 'Backed-Out'].map((status) => {
   341→      const count = placements.filter((p) => p.status === status).length
   342→      return {
   343→        status,
   344→        count,
   345→        percentage: Math.round((count / total) * 100),
   346→      }
   347→    })
   348→  }, [placements])
   349→
   350→  const hasData = placements.length > 0
   351→
   352→  return (
   353→    <motion.div variants={itemVariants}>
   354→      <Card className="rounded-xl">
   355→        <CardContent className="p-5">
   356→          <h3 className="mb-3 text-sm font-semibold text-muted-foreground">Status Pipeline</h3>
   357→          {hasData ? (
   358→            <>
   359→              <div className="flex h-10 overflow-hidden rounded-lg">
   360→                {pipelineData.map((item) => (
   361→                  <motion.div
   362→                    key={item.status}
   363→                    initial={{ width: 0 }}
   364→                    animate={{ width: `${item.percentage}%` }}
   365→                    transition={{ duration: 0.6, ease: 'easeOut', delay: 0.2 }}
   366→                    className={cn(
   367→                      'relative flex items-center justify-center overflow-hidden',
   368→                      PIPELINE_COLORS[item.status],
   369→                      item.percentage < 8 && 'min-w-[2rem]'
   370→                    )}
   371→                  >
   372→                    {item.percentage >= 10 && (
   373→                      <span className="text-xs font-bold text-white drop-shadow-sm">
   374→                        {item.count}
   375→                      </span>
   376→                    )}
   377→                  </motion.div>
   378→                ))}
   379→              </div>
   380→              <div className="mt-2.5 flex">
   381→                {pipelineData.map((item) => (
   382→                  <div
   383→                    key={item.status}
   384→                    className="flex flex-col items-center"
   385→                    style={{ width: `${item.percentage}%`, minWidth: item.percentage < 8 ? '2.5rem' : undefined }}
   386→                  >
   387→                    <span className={cn('text-xs font-semibold', PIPELINE_TEXT_COLORS[item.status])}>
   388→                      {item.status}
   389→                    </span>
   390→                    <span className="text-[10px] text-muted-foreground">{item.percentage}%</span>
   391→                  </div>
   392→                ))}
   393→              </div>
   394→            </>
   395→          ) : (
   396→            <div className="flex h-10 items-center justify-center rounded-lg bg-muted/50">
   397→              <span className="text-xs text-muted-foreground">No data yet</span>
   398→            </div>
   399→          )}
   400→        </CardContent>
   401→      </Card>
   402→    </motion.div>
   403→  )
   404→}
