   174→function formatCurrency(amount: number | null | undefined) {
   175→  if (amount == null) return '—'
   176→  if (amount >= 100000) {
   177→    return `₹${(amount / 100000).toFixed(2)} LPA`
   178→  }
   179→  return `₹${amount.toLocaleString('en-IN')}`
   180→}
   181→
   182→function formatCompactCurrency(amount: number) {
   183→  if (amount >= 10000000) {
   184→    return `₹${(amount / 10000000).toFixed(2)} Cr`
   185→  }
   186→  if (amount >= 100000) {
   187→    return `₹${(amount / 100000).toFixed(1)} L`
   188→  }
   189→  if (amount >= 1000) {
   190→    return `₹${(amount / 1000).toFixed(1)}K`
   191→  }
   192→  return `₹${amount.toFixed(0)}`
   193→}

   195→function formatDate(dateStr: string | null) {
   196→  if (!dateStr) return '—'
   197→  const date = new Date(dateStr)
   198→  return date.toLocaleDateString('en-IN', {
   199→    day: '2-digit',
   200→    month: 'short',
   201→    year: 'numeric',
   202→  })
   203→}

   206→  return `${(firstName?.[0] || '').toUpperCase()}${(lastName?.[0] || '').toUpperCase()}`
   207→}

   209→function exportToCSV(placements: Placement[]) {
   210→  const headers = ['Candidate', 'Client', 'Job Title', 'Recruiter', 'Offered CTC', 'Joined Date', 'Status', 'Commission']
   211→  const rows = placements.map((p) => [
   212→    `${p.candidate.firstName} ${p.candidate.lastName}`,
   213→    p.client?.name || 'N/A',
   214→    p.job?.title || 'N/A',
   215→    p.recruiter?.name || 'N/A',
   216→    p.offeredCTC ?? '',
   217→    p.joinedDate ? formatDate(p.joinedDate) : '',
   218→    p.status,
   219→    p.commission ?? '',
   220→  ])
   221→  const csv = [headers, ...rows].map((row) => row.map((cell) => `"${cell}"`).join(',')).join('\n')
   222→  const blob = new Blob([csv], { type: 'text/csv' })
   223→  const url = URL.createObjectURL(blob)
   224→  const a = document.createElement('a')
   225→  a.href = url
   226→  a.download = `placements-${new Date().toISOString().split('T')[0]}.csv`
   227→  a.click()
   228→  URL.revokeObjectURL(url)
   229→}

   338→// ===== Status Pipeline =====
   339→
   340→function StatusPipeline({ placements }: { placements: Placement[] }) {
   341→  const pipelineData = useMemo(() => {
   342→    const total = placements.length || 1
   343→    return ['Offered', 'Accepted', 'Joined', 'Backed-Out'].map((status) => {
   344→      const count = placements.filter((p) => p.status === status).length
   345→      return {
   346→        status,
   347→        count,
   348→        percentage: Math.round((count / total) * 100),
   349→      }
   350→    })
   351→  }, [placements])
   352→
   353→  const hasData = placements.length > 0
   354→
   355→  return (
   356→    <motion.div variants={itemVariants}>
   357→      <Card className="rounded-xl">
   358→        <CardContent className="p-5">
   359→          <h3 className="mb-3 text-sm font-semibold text-muted-foreground">Status Pipeline</h3>
   360→          {hasData ? (
   361→            <>
   362→              <div className="flex h-10 overflow-hidden rounded-lg">
   363→                {pipelineData.map((item) => (
   364→                  <motion.div
   365→                    key={item.status}
   366→                    initial={{ width: 0 }}
   367→                    animate={{ width: `${item.percentage}%` }}
   368→                    transition={{ duration: 0.6, ease: 'easeOut', delay: 0.2 }}
   369→                    className={cn(
   370→                      'relative flex items-center justify-center overflow-hidden',
   371→                      PIPELINE_COLORS[item.status],
   372→                      item.percentage < 8 && 'min-w-[2rem]'
   373→                    )}
   374→                  >
   375→                    {item.percentage >= 10 && (
   376→                      <span className="text-xs font-bold text-white drop-shadow-sm">
   377→                        {item.count}
   378→                      </span>
   379→                    )}
   380→                  </motion.div>
   381→                ))}
   382→              </div>
   383→              <div className="mt-2.5 flex">
   384→                {pipelineData.map((item) => (
   385→                  <div
   386→                    key={item.status}
   387→                    className="flex flex-col items-center"
   388→                    style={{ width: `${item.percentage}%`, minWidth: item.percentage < 8 ? '2.5rem' : undefined }}
   389→                  >
   390→                    <span className={cn('text-xs font-semibold', PIPELINE_TEXT_COLORS[item.status])}>
   391→                      {item.status}
   392→                    </span>
   393→                    <span className="text-[10px] text-muted-foreground">{item.percentage}%</span>
   394→                  </div>
   395→                ))}
   396→              </div>
   397→            </>
   398→          ) : (
   399→            <div className="flex h-10 items-center justify-center rounded-lg bg-muted/50">
   400→              <span className="text-xs text-muted-foreground">No data yet</span>
   401→            </div>
   402→          )}
   403→        </CardContent>
   404→      </Card>
   405→    </motion.div>
   406→  )
   407→}
