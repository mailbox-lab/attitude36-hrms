   104→function getMonday(d: Date): Date {
   105→  const date = new Date(d.getFullYear(), d.getMonth(), d.getDate())
   106→  const day = date.getDay()
   107→  const diff = day === 0 ? -6 : 1 - day
   108→  date.setDate(date.getDate() + diff)
   109→  date.setHours(0, 0, 0, 0)
   110→  return date
   111→}

   113→function getWeekDays(monday: Date): Date[] {
   114→  const days: Date[] = []
   115→  for (let i = 0; i < 7; i++) {
   116→    const d = new Date(monday)
   117→    d.setDate(monday.getDate() + i)
   118→    days.push(d)
   119→  }
   120→  return days
   121→}

   123→function isSameDay(a: Date, b: Date): boolean {
   124→  return (
   125→    a.getFullYear() === b.getFullYear() &&
   126→    a.getMonth() === b.getMonth() &&
   127→    a.getDate() === b.getDate()
   128→  )
   129→}

   131→function isToday(d: Date): boolean {
   132→  return isSameDay(d, new Date())

   135→function formatTime(date: Date): string {
   136→  let hours = date.getHours()
   137→  const minutes = date.getMinutes()
   138→  const ampm = hours >= 12 ? 'PM' : 'AM'
   139→  hours = hours % 12
   140→  if (hours === 0) hours = 12
   141→  return `${hours}:${minutes.toString().padStart(2, '0')} ${ampm}`
   142→}
   143→
   144→function formatHour(hour: number): string {
   145→ const ampm = hour >= 12 ? 'PM' : 'AM'
   146→  const h = hour % 12 || 12
   147→  return `${h} ${ampm}`
   148→}

   150→function getMonthGrid(year: number, month: number): (Date | null)[] {
   151→  const firstDay = new Date(year, month, 1)
   152→  const lastDay = new Date(year, month + 1, 0)
   153→  const daysInMonth = lastDay.getDate()
   154→
   155→  let startDay = firstDay.getDay() // 0=Sun
   156→  // Adjust for Monday start
   157→  startDay = startDay === 0 ? 6 : startDay - 1
   158→
   159→  const cells: (Date | null)[] = []
   160→
   161→  // Leading empty cells
   162→  for (let i = 0; i < startDay; i++) {
   163→    cells.push(null)
   164→  }
   165→
   166→  // Month days
   167→  for (let d = 1; d <= daysInMonth; d++) {
   168→    cells.push(new Date(year, month, d))
   169→  }
   170→
   171→  // Trailing empty cells to fill 6 rows (42 cells) or at least complete last row
   172→  const remaining = cells.length % 7 === 0 ? 0 : 7 - (cells.length % 7)
   173→  for (let i = 0; i < remaining; i++) {
   174→    cells.push(null)
   175→  }
   176→
   177→  return cells
   178→}

   781→          })}

   783→      </div>
   784→
   785→      {/* Day detail dialog */}
   786→      <DayInterviewsDialog
