   137→function formatDate(dateStr: string) {
   138→  const date = new Date(dateStr)
   139→  return date.toLocaleDateString('en-IN', {
   140→    day: '2-digit',
   141→    month: 'short',
   142→    year: 'numeric',
   143→  })
   144→}
   145→
   146→function formatTime(dateStr: string) {
   147→  const date = new Date(dateStr)
   148→  return date.toLocaleTimeString('en-IN', {
   149→    hour: '2-digit',
   150→    minute: '2-digit',
   151→    hour12: true,
   152→  })
   153→}
   154→
   155→function formatDuration(minutes: number) {
   156→  if (minutes < 60) return `${minutes}m`
   157→  const h = Math.floor(minutes / 60)
   158→  const m = minutes % 60
   159→  return m > 0 ? `${h}h ${m}m` : `${h}h`
   160→}

   414→function FeedbackDialog({
   415→  open,
   416→  onOpenChange,
   417→  interview,
   418→}: {
   419→  open: boolean
   420→  onOpenChange: (open: boolean) => void
   421→  interview: Interview | null
   422→}) {

   427→    defaultValues: {
   428→      feedback: '',
   429→      rating: 3,
   430→    },
   431→  })
   432→
   433→  const mutation = useMutation({
   434→    mutationFn: async (data: FeedbackFormData) => {
   435→      const res = await fetch(`/api/interviews/${interview?.id}`, {
   436→        method: 'PUT',
   437→        headers: { 'Content-Type': 'application/json' },

   439→      })
   440→      if (!res.ok) throw new Error('Failed to save feedback')

   460→      <DialogContent className="sm:max-w-md">
