   201→function getRelativeTime(dateStr: string): string {
   202→  const now = Date.now()
   203→  const then = new Date(dateStr).getTime()
   204→  const diffMs = now - then
   205→  const diffSeconds = Math.floor(diffMs / 1000)
   206→  const diffMinutes = Math.floor(diffSeconds / 60)
   207→  const diffHours = Math.floor(diffMinutes / 60)
   208→  const diffDays = Math.floor(diffHours / 24)
   209→  const diffWeeks = Math.floor(diffDays / 7)
   210→
   211→  if (diffSeconds < 60) return 'Just now'
   212→  if (diffMinutes === 1) return '1 min ago'
   213→  if (diffMinutes < 60) return `${diffMinutes} min ago`
   214→  if (diffHours === 1) return '1 hour ago'
   215→  if (diffHours < 24) return `${diffHours} hours ago`
   216→  if (diffDays === 1) return 'Yesterday'
   217→  if (diffDays < 7) return `${diffDays} days ago`
   218→  if (diffWeeks === 1) return '1 week ago'
   219→  return `${diffWeeks} weeks ago`
   220→}

   222→function isToday(dateStr: string): boolean {
   223→  const d = new Date(dateStr)
   224→  const now = new Date()
   225→  return d.getDate() === now.getDate() &&
   226→    d.getMonth() === now.getMonth() &&
   227→    d.getFullYear() === now.getFullYear()
   228→}
   229→
   230→function isThisWeek(dateStr: string): boolean {
   231→  const d = new Date(dateStr)
   232→  const now = new Date()
   233→  const startOfWeek = new Date(now)
   234→  startOfWeek.setDate(now.getDate() - now.getDay())
   235→  startOfWeek.setHours(0, 0, 0, 0)
   236→  return d >= startOfWeek
   237→}
   238→
   239→function isThisMonth(dateStr: string): boolean {
   240→  const d = new Date(dateStr)
   241→  const now = new Date()
   242→  return d.getMonth() === now.getMonth() && d.getFullYear() === now.getFullYear()
   243→}
   244→
   245→function getDateRangeFilter(dateRange: string, customFrom?: string, customTo?: string): { from?: string; to?: string } {
   246→  if (dateRange === 'today') {
   247→    const today = new Date().toISOString().split('T')[0]
   248→    return { from: today, to: today }
   249→  }
   250→  if (dateRange === 'week') {
   251→    const now = new Date()
   252→    const start = new Date(now)
   253→    start.setDate(now.getDate() - now.getDay())
   254→    start.setHours(0, 0, 0, 0)
   255→    return { from: start.toISOString().split('T')[0], to: now.toISOString().split('T')[0] }
   256→  }
   257→  if (dateRange === 'month' && !customFrom && !customTo) {
   258→    const now = new Date()
   259→    const start = new Date(now.getFullYear(), now.getMonth(), 1)
   260→    return { from: start.toISOString().split('T')[0], to: now.toISOString().split('T')[0] }
   261→  }
   262→  if (dateRange === 'custom' && customFrom && customTo) {
   263→    return { from: customFrom, to: customTo }
   264→  }
   265→  return {}
   266→}

   336→export function NotificationCenter() {
   337→  const { navigate } = useCRMStore()
   338→
   339→  // Filters
   340→  const [entityTypeFilter, setEntityTypeFilter] = useState<string>('all')
   341→  const [actionFilter, setActionFilter] = useState<string>('all')
   342→  const [readFilter, setReadFilter] = useState<'all' | 'unread' | 'read'>('all')
   343→  const [dateRange, setDateRange] = useState<string>('all')
   344→  const [customFrom, setCustomFrom] = useState<string>('')
   345→  const [customTo, setCustomTo] = useState<string>('')
   346→
   347→  // Client-side read tracking
   348→  const [readIds, setReadIds] = useState<Set<string>>(new Set())
   349→
   350→  // Selection for batch actions
   351→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set())
   352→
   353→  // Refs
   354→  const observerRef = useRef<IntersectionObserver | null>(null)
   355→  const loadMoreRef = useRef<HTMLDivElement>(null)
   356→
   357→  // Build filter key for query invalidation
   358→  const filterKey = useMemo(
   359→    () => `${entityTypeFilter}-${actionFilter}-${dateRange}-${customFrom}-${customTo}`,
   360→    [entityTypeFilter, actionFilter, dateRange, customFrom, customTo]
   361→  )
   362→
   363→  // Fetch notifications with infinite query
   364→  const fetchPage = useCallback(async ({ pageParam = 1 }: { pageParam?: number }): Promise<ActivityResponse> => {
   365→    const params = new URLSearchParams()
   366→    params.set('page', String(pageParam))
   367→    params.set('limit', '25')
   368→    if (entityTypeFilter !== 'all') params.set('entityType', entityTypeFilter)
   369→    if (actionFilter !== 'all') params.set('action', actionFilter)
   370→    const res = await fetch(`/api/activity?${params.toString()}`)
   371→    if (!res.ok) throw new Error('Failed to fetch activities')
   372→    return res.json()
   373→  }, [entityTypeFilter, actionFilter])
   374→
   375→  const {
   376→    data,
   377→    isLoading,
   378→    isFetchingNextPage,
   379→    isError,
   380→    hasNextPage,
   381→    fetchNextPage,
   382→  } = useInfiniteQuery({
   383→    queryKey: ['notifications', filterKey],
   384→    queryFn: fetchPage,
   385→    initialPageParam: 1,
   386→    getNextPageParam: (lastPage) => {
   387→      return lastPage.pagination.page < lastPage.pagination.totalPages
   388→        ? lastPage.pagination.page + 1
   389→        : undefined
   390→    },
   391→    staleTime: 30_000,
   392→  })
   393→
   394→  // Flatten pages
   395→  const allActivities = useMemo(() => {
   396→    return data?.pages.flatMap(p => p.data) ?? []
   397→  }, [data])
   398→
   399→  const totalRecords = data?.pages[0]?.pagination.total ?? 0
   400→
   401→  // Stats query (unfiltered)
   402→  const { data: statsData } = useQuery({
   403→    queryKey: ['notifications-stats'],
   404→    queryFn: async () => {
   405→      const res = await fetch('/api/activity?page=1&limit=1000')
   406→      if (!res.ok) throw new Error('Failed')
   407→      const json = await res.json()
   408→      return json.data as ActivityItem[]
   409→    },
   410→    staleTime: 60_000,
   411→  })
   412→
   413→  const stats = useMemo(() => {
   414→    const items = statsData ?? []
   415→    const totalCount = items.length
   416→    const unreadCount = items.filter(a => !readIds.has(a.id)).length
   417→    const todayCount = items.filter(a => isToday(a.createdAt)).length
   418→    return { totalCount, unreadCount, todayCount }
   419→  }, [statsData, readIds])
   420→
   421→  // Available actions based on entity type
   422→  const availableActions = useMemo(() => {
   423→    if (entityTypeFilter === 'all') return Object.keys(ACTION_LABELS)
   424→    return ENTITY_ACTIONS[entityTypeFilter] || []
   425→  }, [entityTypeFilter])
   426→
   427→  // Apply client-side filters (read/unread, date range)
   428→  const filteredActivities = useMemo(() => {
   429→    let filtered = allActivities
   430→
   431→    // Date range filter (client-side since API doesn't support it)
   432→    if (dateRange !== 'all') {
   433→      const range = getDateRangeFilter(dateRange, customFrom, customTo)
   434→      if (range.from && range.to) {
   435→        filtered = filtered.filter(a => {
   436→          const dateStr = a.createdAt.split('T')[0]
   437→          return dateStr >= range.from! && dateStr <= range.to!
   438→        })
   439→      }
   440→    }
   441→
   442→    // Read/unread filter
   443→    if (readFilter === 'unread') {
   444→      filtered = filtered.filter(a => !readIds.has(a.id))
   445→    } else if (readFilter === 'read') {
   446→      filtered = filtered.filter(a => readIds.has(a.id))
   447→    }
   448→
   449→    return filtered
   450→  }, [allActivities, dateRange, customFrom, customTo, readFilter, readIds])
   451→
   452→  // Infinite scroll with IntersectionObserver
   453→  useEffect(() => {
   454→    if (observerRef.current) observerRef.current.disconnect()
   455→    observerRef.current = new IntersectionObserver(
   456→      (entries) => {
   457→        if (entries[0].isIntersecting && hasNextPage && !isFetchingNextPage) {
   458→          fetchNextPage()
   459→        }
   460→      },
   461→      { rootMargin: '200px' }
   462→    )
   463→    if (loadMoreRef.current) observerRef.current.observe(loadMoreRef.current)
   464→    return () => observerRef.current?.disconnect()
   465→  }, [hasNextPage, isFetchingNextPage, fetchNextPage])
   466→
   467→  // Handlers
   468→  const hasActiveFilters = entityTypeFilter !== 'all' || actionFilter !== 'all' || readFilter !== 'all' || dateRange !== 'all'
   469→
   470→  const clearFilters = () => {
   471→    setEntityTypeFilter('all')
   472→    setActionFilter('all')
   473→    setReadFilter('all')
   474→    setDateRange('all')
   475→    setCustomFrom('')
   476→    setCustomTo('')
   477→    setSelectedIds(new Set())
   478→  }
   479→
   480→  const handleMarkAllRead = () => {
   481→    if (allActivities.length === 0) return
   482→    setReadIds(new Set(allActivities.map(a => a.id)))
   483→    toast.success('All notifications marked as read')
   484→  }
   485→
   486→  const handleToggleRead = (id: string) => {
   487→    setReadIds(prev => {
   488→      const next = new Set(prev)
   489→      if (next.has(id)) {
   490→        next.delete(id)
   491→      } else {
   492→        next.add(id)
   493→      }
   494→      return next
   495→    })
   496→  }
   497→
   498→  const handleNotificationClick = (item: ActivityItem) => {
   499→    setReadIds(prev => new Set(prev).add(item.id))
   500→    const config = ENTITY_CONFIG[item.entityType]
   501→    if (config) {
   502→      navigate(config.view)
   503→    }
   504→  }
   505→
   506→  const handleToggleSelect = (id: string) => {
   507→    setSelectedIds(prev => {
   508→      const next = new Set(prev)
   509→      if (next.has(id)) {
   510→        next.delete(id)
   511→      } else {
   512→        next.add(id)
   513→      }
   514→      return next
   515→    })
   516→  }
   517→
   518→  const handleSelectAll = () => {
   519→    if (selectedIds.size === filteredActivities.length) {
   520→      setSelectedIds(new Set())
   521→    } else {
   522→      setSelectedIds(new Set(filteredActivities.map(a => a.id)))
   523→    }
   524→  }
   525→
   526→  const handleBatchMarkRead = () => {
   527→    setReadIds(prev => {
   528→      const next = new Set(prev)
   529→      selectedIds.forEach(id => next.add(id))
   530→      return next
   531→    })
   532→    toast.success(`${selectedIds.size} notification${selectedIds.size !== 1 ? 's' : ''} marked as read`)
   533→    setSelectedIds(new Set())
   534→  }
   535→
   536→  const handleBatchMarkUnread = () => {
   537→    setReadIds(prev => {
   538→      const next = new Set(prev)
   539→      selectedIds.forEach(id => next.delete(id))
   540→      return next
   541→    })
   542→    toast.success(`${selectedIds.size} notification${selectedIds.size !== 1 ? 's' : ''} marked as unread`)
   543→    setSelectedIds(new Set())
   544→  }
   545→
   546→  return (
   547→    <div className="space-y-6">
   548→      {/* Page Header */}
   549→      <motion.div
   550→        initial={{ opacity: 0, y: -10 }}
   551→        animate={{ opacity: 1, y: 0 }}
   552→        transition={{ duration: 0.3 }}
   553→        className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4"
   554→      >
   555→        <div className="flex items-center gap-3">
   556→          <div className="flex h-10 w-10 items-center justify-center rounded-xl bg-amber-100 dark:bg-amber-950 text-amber-600 dark:text-amber-400">
   557→            <Bell className="h-5 w-5" />
