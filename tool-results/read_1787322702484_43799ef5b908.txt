   138→// ===== Helper =====
   139→
   140→function getInitials(firstName: string, lastName: string) {
   141→  return `${firstName.charAt(0)}${lastName.charAt(0)}`.toUpperCase()
   142→}
   143→
   144→function parseSkills(skills: string): string[] {
   145→  return skills
   146→    ? skills
   147→        .split(',')
   148→        .map((s) => s.trim())
   149→        .filter(Boolean)
   150→        .slice(0, 4)
   151→    : []
   152→}

   156→function KanbanCard({
   157→  candidate,
   158→  onView,
   159→  onEdit,
   160→  isSelected,
   161→  onToggleSelect,
   162→}: {
   163→  candidate: Candidate
   164→  onView: () => void
   165→  onEdit: () => void
   166→  isSelected: boolean
   167→  onToggleSelect: () => void
   168→}) {
   169→  const skills = parseSkills(candidate.skills)
   170→  const initials = getInitials(candidate.firstName, candidate.lastName)
   171→
   172→  return (
   173→    <div
   174→      className={`group relative rounded-lg border border-l-4 bg-card p-3.5 shadow-sm transition-all hover:-translate-y-0.5 hover:shadow-md ${
   175→        STATUS_BORDER_COLORS[candidate.status] || 'border-l-gray-500'
   176→      } ${isSelected ? 'ring-2 ring-primary/50 bg-primary/5' : ''}`}
   177→    >
   178→      <div className={`absolute right-2 top-2 z-10 transition-opacity ${isSelected ? 'opacity-100' : 'opacity-0 group-hover:opacity-100'}`}>
   179→        <Checkbox
   180→          checked={isSelected}
   181→          onCheckedChange={(e) => {
   182→            e.stopPropagation()
   183→            onToggleSelect()
   184→          }}
   185→          aria-label={`Select ${candidate.firstName} ${candidate.lastName}`}
   186→          onClick={(e) => e.stopPropagation()}
   187→        />
   188→      </div>
   189→      <div className="flex items-start justify-between gap-2">
   190→        <div className="flex items-center gap-2 min-w-0">
   191→          <Avatar className="h-7 w-7 shrink-0 text-xs">
   192→            <AvatarFallback className="bg-primary/10 text-primary text-[10px]">
   193→              {initials}
   194→            </AvatarFallback>
   195→          </Avatar>
   196→          <div className="min-w-0">
   197→            <button
   198→              onClick={onView}
   199→              className="truncate text-sm font-medium leading-tight hover:underline"
   200→            >
   201→              {candidate.firstName} {candidate.lastName}
   202→            </button>
   203→            {candidate.title && (
   204→              <p className="truncate text-[11px] text-muted-foreground">{candidate.title}</p>
   205→            )}
   206→          </div>
   207→        </div>
   208→        <DropdownMenu>
   209→          <DropdownMenuTrigger asChild>
   210→            <Button
   211→              variant="ghost"
   212→              size="icon"
   213→              className="h-6 w-6 shrink-0 opacity-0 transition-opacity group-hover:opacity-100"
   214→            >
   215→              <MoreHorizontal className="h-3.5 w-3.5" />

   220→              <Eye className="mr-2 h-3.5 w-3.5" /> View Details
   221→            </DropdownMenuItem>

   223→              <Pencil className="mr-2 h-3.5 w-3.5" /> Edit
   224→            </DropdownMenuItem>
   225→          </DropdownMenuContent>
   226→        </DropdownMenu>
   227→      </div>
   228→      {skills.length > 0 && (
   229→        <div className="mt-2 flex flex-wrap gap-1">
   230→          {skills.map((skill) => (
   231→            <span
   232→              key={skill}
   233→              className="inline-flex items-center rounded bg-muted px-1.5 py-0.5 text-[10px] font-medium text-muted-foreground"
   234→            >
   235→              {skill}
   236→            </span>
   237→          ))}
   238→        </div>
   239→      )}
   240→      {(candidate.location || candidate.experience != null) && (
   241→        <div className="mt-2 flex items-center gap-3 text-[10px] text-muted-foreground">
   242→          {candidate.experience != null && (
   243→            <span>{candidate.experience} yrs exp</span>
   244→          )}
   245→          {candidate.location && (
   246→            <span className="flex items-center gap-0.5">
   247→              <MapPin className="h-2.5 w-2.5" /> {candidate.location}
   248→            </span>
   249→          )}
   250→        </div>
   251→      )}
   252→    </div>
   253→  )
   254→}

   307→        borderTopColor:
   308→          status === 'New'
   309→            ? '#10b981'

   345→export function CandidatesPage() {
   346→  const navigate = useCRMStore((s) => s.navigate)
   347→  const selectedId = useCRMStore((s) => s.selectedId)
   348→  const candidateFilter = useCRMStore((s) => s.candidateFilter)
   349→  const setCandidateFilter = useCRMStore((s) => s.setCandidateFilter)
   350→  const queryClient = useQueryClient()
   351→
   352→  const [addDialogOpen, setAddDialogOpen] = useState(false)
   353→  const [editCandidate, setEditCandidate] = useState<Candidate | null>(null)
   354→  const [activeTab, setActiveTab] = useState('list')
   355→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set())
   356→  const [compareIds, setCompareIds] = useState<string[] | null>(null)
   357→
   358→  const selectedIdsArray = useMemo(() => Array.from(selectedIds), [selectedIds])
   359→
   360→  const toggleSelect = useCallback((id: string) => {
   361→    setSelectedIds((prev) => {
   362→      const next = new Set(prev)
   363→      if (next.has(id)) {
   364→        next.delete(id)
   365→      } else {
   366→        next.add(id)
   367→      }
   368→      return next
   369→    })
   370→  }, [])
   371→
   372→  const deselectAll = useCallback(() => {
   373→    setSelectedIds(new Set())
   374→  }, [])
   375→
   376→  function handleBulkSuccess() {
   377→    setSelectedIds(new Set())
   378→    queryClient.invalidateQueries({ queryKey: ['candidates'] })
   379→  }
   380→
   381→  // Fetch candidates with filters
   382→  const { data, isLoading, error } = useQuery<{
   383→    data: Candidate[]
   384→    pagination: { page: number; limit: number; total: number; totalPages: number }
   385→  }>({
   386→    queryKey: ['candidates', candidateFilter],
   387→    queryFn: async () => {
   388→      const params = new URLSearchParams()
   389→      if (candidateFilter.search) params.set('search', candidateFilter.search)
   390→      if (candidateFilter.status && candidateFilter.status !== 'All')
   391→        params.set('status', candidateFilter.status)
   392→      if (candidateFilter.source && candidateFilter.source !== 'All')
   393→        params.set('source', candidateFilter.source)
   394→      const res = await fetch(`/api/candidates?${params.toString()}`)
   395→      if (!res.ok) throw new Error('Failed to fetch candidates')
   396→      return res.json()
   397→    },
   398→  })
   399→
   400→  const candidates = data?.data || []
   401→
   402→  const selectAll = useCallback(() => {
   403→    setSelectedIds(new Set(candidates.map((c) => c.id)))
   404→  }, [candidates])
   405→
   406→  const isAllSelected = candidates.length > 0 && candidates.every((c) => selectedIds.has(c.id))
   407→  const isSomeSelected = !isAllSelected && candidates.some((c) => selectedIds.has(c.id))
   408→
   409→  // Quick status change mutation
   410→  const updateStatusMutation = useMutation({
   411→    mutationFn: async ({ id, status }: { id: string; status: string }) => {
   412→      const res = await fetch(`/api/candidates/${id}`, {
   413→        method: 'PUT',
   414→        headers: { 'Content-Type': 'application/json' },
   415→        body: JSON.stringify({ status }),
   416→      })
   417→      if (!res.ok) throw new Error('Failed to update status')
   418→      return res.json()
   419→    },
   420→    onSuccess: () => {
   421→      queryClient.invalidateQueries({ queryKey: ['candidates'] })
   422→      toast.success('Candidate status updated')
   423→    },
   424→    onError: () => {
   425→      toast.error('Failed to update candidate status')
   426→    },
   427→  })
   428→
   429→  // Delete mutation
   430→  const deleteMutation = useMutation({
   431→    mutationFn: async (id: string) => {
   432→      const res = await fetch(`/api/candidates/${id}`, { method: 'DELETE' })
   433→      if (!res.ok) throw new Error('Failed to delete candidate')
   434→      return res.json()
   435→    },
   436→    onSuccess: () => {
   437→      queryClient.invalidateQueries({ queryKey: ['candidates'] })
   438→      toast.success('Candidate deleted successfully')
   439→    },
   440→    onError: () => {
   441→      toast.error('Failed to delete candidate')
   442→    },
   443→  })
   444→
   445→  // Group candidates by status for Kanban
   446→  const kanbanData = useMemo(() => {
   447→    const groups: Record<string, Candidate[]> = {}
   448→    ALL_STATUSES.forEach((s) => {
   449→      groups[s] = []
   450→    })
   451→    candidates.forEach((c) => {
   452→      if (groups[c.status]) {
   453→        groups[c.status].push(c)
   454→      } else {
   455→        groups['New'].push(c)
   456→      }
   457→    })
   458→    return groups
   459→  }, [candidates])
   460→
   461→  function handleViewCandidate(id: string) {
   462→    navigate('candidate-detail', id)
   463→  }
   464→
   465→  function handleEditCandidate(candidate: Candidate) {
   466→    setEditCandidate(candidate)
   467→    setAddDialogOpen(true)
   468→  }
   469→
   470→  function handleCloseDialog(open: boolean) {
   471→    setAddDialogOpen(open)
   472→    if (!open) setEditCandidate(null)
   473→  }
   474→
   475→  function handleCompare() {
   476→    const ids = Array.from(selectedIds).slice(0, 4)
   477→    setCompareIds(ids)
   478→  }
   479→
   480→  function handleCloseComparison() {
   481→    setCompareIds(null)
   482→  }
   483→
   484→  // If a candidate is selected, show detail view
   485→  if (selectedId) {
   486→    return <CandidateDetail candidateId={selectedId} />
   487→  }
   488→
   489→  return (
   490→    <div className="flex flex-1 flex-col gap-6 p-4 md:p-6">
   491→      {/* Header */}
   492→      <div className="space-y-3">
   493→        <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   494→          <div className="flex items-center gap-3">
   495→            <div className="flex h-10 w-10 items-center justify-center rounded-xl bg-emerald-100 dark:bg-emerald-950">
   496→              <Users className="h-5 w-5 text-emerald-600 dark:text-emerald-400" />
   497→            </div>
   498→            <div>
   499→              <h1 className="text-2xl font-bold tracking-tight md:text-3xl">Candidates</h1>
   500→              <p className="mt-1 text-sm text-muted-foreground">
   501→                Manage and track your candidate pipeline
   502→                {data && (
   503→                  <span className="ml-1 font-medium text-foreground">
   504→                    ({data.pagination.total} total)
   505→                  </span>
   506→                )}
   507→              </p>
   508→            </div>
   509→          </div>
   510→        <div className="flex gap-2">
   511→          <Button
   512→            variant="outline"
   513→            size="sm"
   514→            className="gap-1.5"
   515→            onClick={() => {
   516→              const csvData = candidates.map((c) => ({
   517→                Name: `${c.firstName} ${c.lastName}`,
   518→                Email: c.email || '',
   519→                Phone: c.phone || '',
   520→                Title: c.title || '',
   521→                Status: c.status,
   522→                Source: c.source || '',
   523→                Experience: c.experience ?? '',
   524→                Location: c.location || '',
   525→                Skills: c.skills,
   526→                'Current CTC': c.currentCTC ?? '',
   527→                'Expected CTC': c.expectedCTC ?? '',
   528→                'Notice Period': c.noticePeriod ?? '',
   529→                'Job Title': c.job?.title || '',
   530→              }))
   531→              exportToCSV(csvData, 'candidates')
   532→            }}
   533→          >
   534→            <Download className="h-4 w-4" />
   535→            Export CSV
   536→          </Button>
   537→          <Button onClick={() => setAddDialogOpen(true)} className="bg-gradient-to-r from-primary to-primary/80 shadow-md hover:shadow-lg">
   538→            <Plus className="mr-2 h-4 w-4" />
   539→            {selectedIds.size > 0 ? `Add Candidate (${selectedIds.size} selected)` : 'Add Candidate'}
   540→          </Button>
   541→          {candidates.length > 0 && (
   542→            <div className="flex items-center gap-2">
   543→              <Checkbox
   544→                checked={isAllSelected}
   545→                ref={(el) => {
   546→                  if (el) {
   547→                    (el as unknown as { indeterminate: boolean }).indeterminate = isSomeSelected
   548→                  }
   549→                }}
   550→                onCheckedChange={(checked) => {
   551→                  if (checked) selectAll()
   552→                  else deselectAll()
   553→                }}
   554→                aria-label="Select all candidates"
   555→              />
   556→              <span className="hidden text-xs text-muted-foreground sm:inline">Select All</span>
   557→            </div>
   558→          )}
   559→        </div>
   560→      </div>
   561→      <div className="h-1 w-16 rounded-full bg-gradient-to-r from-emerald-400 to-teal-400" />
   562→      </div>
   563→
   564→      {/* Filters */}
   565→      <div className="flex flex-col gap-3 rounded-lg bg-muted/50 p-2 sm:flex-row sm:items-center">
   566→        <div className="relative flex-1 sm:max-w-xs">
   567→          <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   568→          <Input
   569→            placeholder="Search candidates..."
   570→            className="border-0 bg-background pl-9 shadow-sm"
   571→            value={candidateFilter.search}
   572→            onChange={(e) => setCandidateFilter({ search: e.target.value })}
   573→          />
   574→        </div>
   575→        <Select
   576→          value={candidateFilter.status}
   577→          onValueChange={(val) => setCandidateFilter({ status: val })}
   578→        >
   579→          <SelectTrigger className="w-full border-0 bg-background shadow-sm sm:w-[160px]">
   580→            <SelectValue placeholder="Status" />
   581→          </SelectTrigger>
   582→          <SelectContent>
