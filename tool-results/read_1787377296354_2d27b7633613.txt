   355→// ===== Status Toggle =====
   356→
   357→function StatusToggle({
   358→  employee,
   359→}: {
   360→  employee: Employee
   361→}) {
   362→  const queryClient = useQueryClient()
   363→  const { role: myRole, employeeId: myId } = useAuth()
   364→
   365→  // Only Founder/CoFounder can toggle status
   366→  const canToggle = myRole === 'FOUNDER' || myRole === 'COFOUNDER'
   367→  // Can't deactivate yourself
   368→  const isSelf = myId === employee.id
   369→
   370→  const toggleMutation = useMutation({
   371→    mutationFn: async () => {
   372→      const res = await fetch(`/api/employees/${employee.id}`, {
   373→        method: 'PUT',
   374→        headers: { 'Content-Type': 'application/json' },
   375→        body: JSON.stringify({ isActive: !employee.isActive }),
   376→      })
   377→      if (!res.ok) throw new Error('Failed to update status')
   378→      return res.json()
   379→    },
   380→    onSuccess: () => {
   381→      queryClient.invalidateQueries({ queryKey: ['employees'] })
   382→      toast.success(`Employee ${employee.isActive ? 'deactivated' : 'activated'} successfully`)
   383→    },
   384→    onError: () => {
   385→      toast.error('Failed to update employee status')
   386→    },
   387→  })
   388→
   389→  if (!canToggle || isSelf) {
   390→    return (
   391→      <Badge
   392→        className={`text-[10px] ${
   393→          employee.isActive
   394→            ? 'bg-green-100 text-green-700 dark:bg-green-950 dark:text-green-400'
   395→            : 'bg-red-100 text-red-700 dark:bg-red-950 dark:text-red-400'
   396→        }`}
   397→      >
   398→        {employee.isActive ? 'Active' : 'Inactive'}
   399→      </Badge>
   400→    )
   401→  }
   402→
   403→  return (
   404→    <div
   405→      className="flex items-center gap-2 cursor-pointer group"
   406→      onClick={() => toggleMutation.mutate()}
   407→    >
   408→      <Switch
   409→        checked={employee.isActive}
   410→        disabled={toggleMutation.isPending}
   411→        className="scale-75"
   412→    />
   413→      <span className={`text-xs font-medium ${employee.isActive ? 'text-green-600 dark:text-green-400' : 'text-red-500'}`}>
   414→        {toggleMutation.isPending ? '...' : employee.isActive ? 'Active' : 'Inactive'}
   415→      </span>
   416→    </div>
   417→  )
   418→}
