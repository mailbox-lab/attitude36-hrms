   173→function formatCurrency(amount: number | null | undefined) {
   174→  if (amount == null) return '—'
   175→  if (amount >= 100000) {
   176→    return `₹${(amount / 100000).toFixed(2)} LPA`
   177→  }
   178→  return `₹${amount.toLocaleString('en-IN')}`
   179→}
   180→
   181→function formatCompactCurrency(amount: number) {
   182→  if (amount >= 10000000) {
   183→    return `₹${(amount / 10000000).toFixed(2)} Cr`
   184→  }
   185→  if (amount >= 100000) {
   186→    return `₹${(amount / 100000).toFixed(1)} L`
   187→  }
   188→  if (amount >= 1000) {
   189→    return `₹${(amount / 1000).toFixed(1)}K`
   190→  }
   191→  return `₹${amount.toFixed(0)}`
   192→}

   194→function formatDate(dateStr: string | null) {
   195→  if (!dateStr) return '—'
   196→  const date = new Date(dateStr)
   197→  return date.toLocaleDateString('en-IN', {
   198→    day: '2-digit',
   199→    month: 'short',
   200→    year: 'numeric',
   201→  })
   202→}

   208→function exportToCSV(placements: Placement[]) {
   209→  const headers = ['Candidate', 'Client', 'Job Title', 'Recruiter', 'Offered CTC', 'Joined Date', 'Status', 'Commission']
   210→  const rows = placements.map((p) => [
   211→    `${p.candidate.firstName} ${p.candidate.lastName}`,
   212→    p.client?.name || 'N/A',
   213→    p.job?.title || 'N/A',
   214→    p.recruiter?.name || 'N/A',
   215→    p.offeredCTC ?? '',
   216→    p.joinedDate ? formatDate(p.joinedDate) : '',
   217→    p.status,
   218→    p.commission ?? '',
   219→  ])
   220→  const csv = [headers, ...rows].map((row) => row.map((cell) => `"${cell}"`).join(',')).join('\n')
   221→  const blob = new Blob([csv], { type: 'text/csv' })
   222→  const url = URL.createObjectURL(blob)
   223→  const a = document.createElement('a')
   224→  a.href = url
   225→  a.download = `placements-${new Date().toISOString().split('T')[0]}.csv`
   226→  a.click()
   227→  URL.revokeObjectURL(url)
   228→}

   337→// ===== Status Pipeline =====

   339→function StatusPipeline({ placements }: { placements: Placement[] }) {
   340→  const pipelineData = useMemo(() => {
   341→    const total = placements.length || 1
   342→    return ['Offered', 'Accepted', 'Joined', 'Backed-Out'].map((status) => {
   343→      const count = placements.filter((p) => p.status === status).length
   344→      return {
   345→        status,
   346→        count,
   347→        percentage: Math.round((count / total) * 100),
   348→      }
   349→    })
   350→  }, [placements])
   351→
   352→  const hasData = placements.length > 0
   353→
   354→  return (
   355→    <motion.div variants={itemVariants}>
   356→      <Card className="rounded-xl">
   357→        <CardContent className="p-5">
   358→          <h3 className="mb-3 text-sm font-semibold text-muted-foreground">Status Pipeline</h3>
   359→          {hasData ? (
   360→            <>
   361→              <div className="flex h-10 overflow-hidden rounded-lg">
   362→                {pipelineData.map((item) => (
   363→                  <motion.div
   364→                    key={item.status}
   365→                    initial={{ width: 0 }}
   366→                    animate={{ width: `${item.percentage}%` }}
   367→                    transition={{ duration: 0.6, ease: 'easeOut', delay: 0.2 }}
   368→                    className={cn(
   369→                      'relative flex items-center justify-center overflow-hidden',
   370→                      PIPELINE_COLORS[item.status],
   371→                      item.percentage < 8 && 'min-w-[2rem]'
   372→                    )}
   373→                  >
   374→                    {item.percentage >= 10 && (
   375→                      <span className="text-xs font-bold text-white drop-shadow-sm">
   376→                        {item.count}
   377→                      </span>
   378→                    )}
   379→                  </motion.div>
   380→                ))}
   381→              </div>
   382→              <div className="mt-2.5 flex">
   383→                {pipelineData.map((item) => (
   384→                  <div
   385→                    key={item.status}
   386→                    className="flex flex-col items-center"
   387→                    style={{ width: `${item.percentage}%`, minWidth: item.percentage < 8 ? '2.5rem' : undefined }}
   388→                  >
   389→                    <span className={cn('text-xs font-semibold', PIPELINE_TEXT_COLORS[item.status])}>
   390→                      {item.status}
   391→                    </span>
   392→                    <span className="text-[10px] text-muted-foreground">{item.percentage}%</span>
   393→                  </div>
   394→                ))}
   395→              </div>
   396→            </>
   397→          ) : (
   398→            <div className="flex h-10 items-center justify-center rounded-lg bg-muted/50">
   399→              <span className="text-xs text-muted-foreground">No data yet</span>
   400→            </div>
   401→          )}
   402→        </CardContent>
   403→      </Card>
   404→    </motion.div>
   405→  )
   406→}
