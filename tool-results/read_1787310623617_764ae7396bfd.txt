    45→// ===================== Profile Tab =====================
    46→function ProfileTab() {
    47→  const [name, setName] = useState('')
    48→  const [email, setEmail] = useState('')
    49→  const [phone, setPhone] = useState('')
    50→  const [role, setRole] = useState('')
    51→  const [department, setDepartment] = useState('')
    52→  const [saving, setSaving] = useState(false)
    53→
    54→  useEffect(() => {
    55→    async function loadEmployee() {
    56→      try {
    57→        const res = await fetch('/api/employees')
    58→        if (res.ok) {
    59→          const json = await res.json()
    60→          const employees = json.data || json
    61→          if (Array.isArray(employees) && employees.length > 0) {
    62→            const emp = employees[0]
    63→            setName(emp.name || '')
    64→            setEmail(emp.email || '')
    65→            setPhone(emp.phone || '')
    66→            setRole(emp.role || '')
    67→            setDepartment(emp.department || '')
    68→          }
    69→        }
    70→      } catch {
    71→        // silently fail
    72→      }
    73→    }
    74→    loadEmployee()
    75→  }, [])
    76→
    77→  const getInitials = (name: string) => {
    78→    return name
    79→      .split(' ')
    80→      .map((n) => n[0])
    81→      .join('')
    82→      .toUpperCase()
    83→      .slice(0, 2)
    84→  }
    85→
    86→  const handleSave = async () => {
    87→    setSaving(true)
    88→    // Simulate save with a small delay
    89→    await new Promise((resolve) => setTimeout(resolve, 800))
    90→    setSaving(false)
    91→    toast.success('Profile updated successfully')
    92→  }
    93→
    94→  return (
    95→    <div className="space-y-6">
    96→      <Card>
    97→        <CardHeader>
    98→          <CardTitle className="text-lg">Profile Information</CardTitle>
    99→          <CardDescription>Manage your account details and profile information.</CardDescription>
   100→        </CardHeader>
   101→        <CardContent>
   102→          <div className="flex flex-col sm:flex-row gap-6">
   103→            {/* Avatar Section */}
   104→            <div className="flex flex-col items-center gap-3">
   105→              <Avatar className="h-24 w-24">
   106→                <AvatarFallback className="bg-primary/10 text-primary text-2xl font-bold">
   107→                  {name ? getInitials(name) : 'RP'}
   108→                </AvatarFallback>
   109→              </Avatar>
   110→              <Badge variant="secondary" className="text-xs">
   111→                {role || 'Recruiter'}
   112→              </Badge>
   113→            </div>
   114→
   115→            {/* Form Fields */}
   116→            <div className="flex-1 grid grid-cols-1 md:grid-cols-2 gap-4">
   117→              <div className="space-y-2">
   118→                <Label htmlFor="name">Full Name</Label>
   119→                <Input
   120→                  id="name"
   121→                  value={name}
   122→                  onChange={(e) => setName(e.target.value)}
   123→                  placeholder="Enter your name"
   124→                />
   125→              </div>
   126→              <div className="space-y-2">
   127→                <Label htmlFor="email">Email Address</Label>
   128→                <Input
   129→                  id="email"
   130→                  type="email"
   131→                  value={email}
   132→                  onChange={(e) => setEmail(e.target.value)}
   133→                  placeholder="Enter your email"
   134→                />
   135→              </div>

   146→                <Label htmlFor="role">Role</Label>
   147→                <Input
   148→                  id="role"
   149→                  value={role}

   165→          <Separator className="my-6" />
   166→          <div className="flex justify-end">
   167→            <Button onClick={handleSave} disabled={saving}>

   170→                  <Loader2 className="h-4 w-4 mr-2 animate-spin" />

   187→// ===================== Appearance Tab =====================
   188→function AppearanceTab() {
   189→  const { theme, setTheme } = useTheme()
   190→  const [compactMode, setCompactMode] = useState(() => {
   191→    if (typeof window === 'undefined') return false
   192→    const cm = localStorage.getItem('recruitpro-compact-mode')
   193→    return cm === null ? false : cm === 'true'
   194→  })
   195→  const [sidebarDefault, setSidebarDefault] = useState(() => {
   196→    if (typeof window === 'undefined') return true
   197→    const sd = localStorage.getItem('recruitpro-sidebar-default')
   198→    return sd === null ? true : sd === 'true'
   199→  })
   200→
   201→  const handleCompactChange = (checked: boolean) => {
   202→    setCompactMode(checked)
   203→    localStorage.setItem('recruitpro-compact-mode', String(checked))
   204→    toast.success(checked ? 'Compact mode enabled' : 'Compact mode disabled')
   205→  }
   206→
   207→  const handleSidebarDefault = (checked: boolean) => {
   208→    setSidebarDefault(checked)
   209→    localStorage.setItem('recruitpro-sidebar-default', String(checked))
   210→    toast.success(checked ? 'Sidebar will default to expanded' : 'Sidebar will default to collapsed')
   211→  }
   212→
   213→  const themeOptions = [
   214→    { value: 'light', label: 'Light', icon: Sun },
   215→    { value: 'dark', label: 'Dark', icon: Moon },
   216→    { value: 'system', label: 'System', icon: Monitor },
   217→  ] as const
   218→
   219→  return (
   220→    <div className="space-y-6">
   221→      {/* Theme Selection */}
   222→      <Card>
   223→        <CardHeader>
   224→          <CardTitle className="text-lg">Theme</CardTitle>
   225→          <CardDescription>Select your preferred color theme for the application.</CardDescription>
   226→        </CardHeader>
   227→        <CardContent>
   228→          <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   229→            {themeOptions.map((option) => {
   230→              const Icon = option.icon
   231→              const isActive = theme === option.value
   232→              return (
   233→                <button
   234→                  key={option.value}
   235→                  onClick={() => setTheme(option.value)}
   236→                  className={cn(
   237→                    'relative flex flex-col items-center gap-3 rounded-xl border-2 p-6 transition-all hover:bg-accent/50 cursor-pointer',
   238→                    isActive
   239→                      ? 'border-primary bg-primary/5 shadow-sm'
   240→                      : 'border-muted hover:border-muted-foreground/30'
   241→                  )}
   242→                >
   243→                  {isActive && (
   244→                    <div className="absolute top-2 right-2">
   245→                      <div className="h-5 w-5 rounded-full bg-primary flex items-center justify-center">
   246→                        <Check className="h-3 w-3 text-primary-foreground" />
   247→                      </div>
   248→                    </div>
   249→                  )}
   250→                  <Icon className={cn('h-8 w-8', isActive ? 'text-primary' : 'text-muted-foreground')} />
   251→                  <span className={cn('text-sm font-medium', isActive ? 'text-primary' : 'text-foreground')}>
   252→                    {option.label}
   253→                  </span>
   254→                </button>
   255→              )
   256→            })}
   257→          </div>
   258→        </CardContent>
   259→      </Card>
   260→
   261→      {/* Compact Mode */}
   262→      <Card>
   263→        <CardHeader>
   264→          <CardTitle className="text-lg">Display Options</CardTitle>
   265→          <CardDescription>Customize the display density and layout preferences.</CardDescription>
   266→        </CardHeader>
   267→        <CardContent className="space-y-6">
   268→          <div className="flex items-center justify-between">
   269→            <div className="space-y-0.5">
   270→              <Label className="text-sm font-medium">Compact Mode</Label>
   271→              <p className="text-xs text-muted-foreground">Reduce spacing and padding for a denser layout.</p>
   272→            </div>
   273→            <Switch checked={compactMode} onCheckedChange={handleCompactChange} />
   274→          </div>
   275→          <Separator />
   276→          <div className="flex items-center justify-between">
   277→            <div className="space-y-0.5">
   278→              <Label className="text-sm font-medium">Sidebar Default State</Label>
   279→              <p className="text-xs text-muted-foreground">
   280→                {sidebarDefault ? 'Sidebar starts expanded on page load.' : 'Sidebar starts collapsed on page load.'}
   281→              </p>
   282→            </div>
   283→            <Switch checked={sidebarDefault} onCheckedChange={handleSidebarDefault} />
   284→          </div>
   285→        </CardContent>
   286→      </Card>
   287→    </div>
   288→  )
   289→}

   291→// ===================== Notifications Tab =====================
   292→function NotificationsTab() {
   293→  const lsBool = (key: string, fallback: boolean) => {
   294→    if (typeof window === 'undefined') return fallback
   295→    const val = localStorage.getItem(key)
   296→    return val === null ? fallback : val === 'true'
   297→  }
   298→
   299→  const [emailNotifications, setEmailNotifications] = useState(() => lsBool('recruitpro-notif-email', true))
   300→  const [interviewReminders, setInterviewReminders] = useState(() => lsBool('recruitpro-notif-interview', true))
   301→  const [leaveAlerts, setLeaveAlerts] = useState(() => lsBool('recruitpro-notif-leave', true))
   302→  const [placementUpdates, setPlacementUpdates] = useState(() => lsBool('recruitpro-notif-placement', true))
   303→
   304→  const toggleSetting = (
   305→    key: string,
   306→    value: boolean,
   307→    setter: (v: boolean) => void,
   308→    label: string
   309→  ) => {
   310→    setter(value)
   311→    localStorage.setItem(key, String(value))
   312→    toast.success(`${label} ${value ? 'enabled' : 'disabled'}`)
   313→  }
   314→
   315→  const notificationItems = [
   316→    {
   317→      label: 'Email Notifications',
   318→      description: 'Receive email notifications for important events and updates.',
   319→      value: emailNotifications,
   320→      setter: (v: boolean) => toggleSetting('recruitpro-notif-email', v, setEmailNotifications, 'Email Notifications'),
   321→    },
   322→    {
   323→      label: 'Interview Reminders',
   324→      description: 'Get reminded about upcoming interviews with candidates.',
   325→      value: interviewReminders,
   326→      setter: (v: boolean) => toggleSetting('recruitpro-notif-interview', v, setInterviewReminders, 'Interview Reminders'),
   327→    },
   328→    {
   329→      label: 'Leave Request Alerts',
   330→      description: 'Receive alerts when new leave requests are submitted.',
   331→      value: leaveAlerts,
   332→      setter: (v: boolean) => toggleSetting('recruitpro-notif-leave', v, setLeaveAlerts, 'Leave Request Alerts'),
   333→    },
   334→    {
   335→      label: 'Placement Updates',
   336→      description: 'Get notified about placement status changes and milestones.',
   337→      value: placementUpdates,
   338→      setter: (v: boolean) => toggleSetting('recruitpro-notif-placement', v, setPlacementUpdates, 'Placement Updates'),
   339→    },
   340→  ]
   341→
   342→  return (
   343→    <div className="space-y-6">
   344→      <Card>
   345→        <CardHeader>
   346→          <CardTitle className="text-lg">Notification Preferences</CardTitle>
   347→          <CardDescription>Choose which notifications you would like to receive.</CardDescription>
   348→        </CardHeader>
   349→        <CardContent className="space-y-6">
   350→          {notificationItems.map((item, index) => (
   351→            <div key={item.label}>
   352→              <div className="flex items-center justify-between">
   353→                <div className="space-y-0.5 pr-4">
   354→                  <Label className="text-sm font-medium">{item.label}</Label>
   355→                  <p className="text-xs text-muted-foreground">{item.description}</p>
   356→                </div>
   357→                <Switch checked={item.value} onCheckedChange={item.setter} />
   358→              </div>
   359→              {index < notificationItems.length - 1 && <Separator className="mt-6" />}
   360→            </div>
   361→          ))}
   362→        </CardContent>
   363→      </Card>
   364→    </div>
   365→  )
   366→}

   368→// ===================== Data Management Tab =====================
   369→function DataManagementTab() {
   370→  const [stats, setStats] = useState({
   371→    totalCandidates: 0,
   372→    totalClients: 0,
   373→    totalJobs: 0,
   374→    totalEmployees: 0,
   375→  })
   376→  const [statsLoading, setStatsLoading] = useState(true)
   377→  const [exporting, setExporting] = useState(false)
   378→  const [clearing, setClearing] = useState(false)
   379→  const [reseeding, setReseeding] = useState(false)
   380→
   381→  const fetchStats = useCallback(async () => {
   382→    try {
   383→      setStatsLoading(true)
   384→      const [dashRes, clientRes, empRes] = await Promise.all([
   385→        fetch('/api/dashboard'),
   386→        fetch('/api/clients'),
   387→        fetch('/api/employees?activeOnly=false'),
   388→      ])
   389→
   390→      const dashData = dashRes.ok ? await dashRes.json() : {}
   391→      const clientData = clientRes.ok ? await clientRes.json() : { data: [] }
   392→      const empData = empRes.ok ? await empRes.json() : { data: [] }
   393→
   394→      const clientList = clientData.data || clientData || []
   395→      const empList = empData.data || empData || []
   396→
   397→      // Total clients from API (all, not just active)
   398→      const totalClients = Array.isArray(clientList) ? clientList.length : 0
   399→      const totalEmployees = Array.isArray(empList) ? empList.length : 0
   400→
   401→      // Total jobs: we need open + closed. The dashboard gives us openJobs.
   402→      // For total, we fetch from the jobs API
   403→      const jobsRes = await fetch('/api/jobs')
   404→      const jobsData = jobsRes.ok ? await jobsRes.json() : { data: [] }
   405→      const jobList = jobsData.data || jobsData || []
   406→      const totalJobs = Array.isArray(jobList) ? jobList.length : 0
   407→
   408→      setStats({
   409→        totalCandidates: dashData.totalCandidates || 0,
   410→        totalClients,
   411→        totalJobs,
   412→        totalEmployees,
   413→      })
   414→    } catch {
   415→      // silently fail
   416→    } finally {
   417→      setStatsLoading(false)
   418→    }
   419→  }, [])
   420→
   421→  useEffect(() => {
   422→    fetchStats()
   423→  }, [fetchStats])
   424→
   425→  const handleExport = async () => {
   426→    setExporting(true)
   427→    try {
   428→      const res = await fetch('/api/candidates')
   429→      if (!res.ok) throw new Error('Failed to fetch candidates')
   430→      const json = await res.json()
   431→      const candidates = json.data || json || []
   432→
   433→      if (candidates.length === 0) {
   434→        toast.error('No candidates to export')
   435→        setExporting(false)
   436→        return
   437→      }
   438→
   439→      // Build CSV
   440→      const headers = ['Name', 'Email', 'Phone', 'Title', 'Location', 'Experience', 'Status', 'Source', 'Rating']
   441→      const rows = candidates.map((c: Record<string, unknown>) => [
   442→        `${c.firstName || ''} ${c.lastName || ''}`.trim(),
   443→        c.email || '',
   444→        c.phone || '',
   445→        c.title || '',
   446→        c.location || '',
   447→        c.experience ?? '',
   448→        c.status || '',
   449→        c.source || '',
   450→        c.rating ?? '',
   451→      ])
   452→
   453→      const csvContent = [headers, ...rows]
   454→        .map((row) => row.map((cell) => `"${String(cell).replace(/"/g, '""')}"`).join(','))
   455→        .join('\n')
   456→
   457→      const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' })
   458→      const url = URL.createObjectURL(blob)
   459→      const link = document.createElement('a')
   460→      link.href = url
   461→      link.download = `recruitpro-candidates-${new Date().toISOString().split('T')[0]}.csv`
   462→      link.click()
   463→      URL.revokeObjectURL(url)
   464→
   465→      toast.success(`Exported ${candidates.length} candidates to CSV`)
   466→    } catch {
   467→      toast.error('Failed to export data')
   468→    } finally {
   469→      setExporting(false)
   470→    }
   471→  }
   472→
   473→  const handleClearAll = async () => {
   474→    setClearing(true)
   475→    try {
   476→      // Delete all data by model type
   477→      const models = ['placements', 'interviews', 'leave', 'attendance', 'candidates', 'jobs', 'clients', 'employees']
   478→      await Promise.all(
   479→        models.map(async (model) => {
   480→          // Fetch all IDs first, then delete individually
   481→          try {
   482→            const res = await fetch(`/api/${model}`)
   483→            if (!res.ok) return
   484→            const json = await res.json()
   485→            const items = json.data || json || []
   486→            if (Array.isArray(items)) {
   487→              await Promise.all(
   488→                items.map((item: Record<string, unknown>) =>
   489→                  fetch(`/api/${model}/${item.id}`, { method: 'DELETE' }).catch(() => {})
   490→                )
   491→              )
   492→            }
   493→          } catch {
   494→            // continue
   495→          }
   496→        })
   497→      )
   498→      toast.success('All data has been cleared successfully')
   499→      fetchStats()
   500→    } catch {
   501→      toast.error('Failed to clear data')
   502→    } finally {
   503→      setClearing(false)
   504→    }
   505→  }
   506→
   507→  const handleReseed = async () => {
   508→    setReseeding(true)
   509→    try {
   510→      const res = await fetch('/api/seed', { method: 'POST' })
   511→      if (res.ok) {
   512→        toast.success('Sample data has been reseeded successfully')
   513→        fetchStats()
   514→      } else {
   515→        toast.error('Failed to reseed data')
   516→      }
   517→    } catch {
   518→      toast.error('Failed to reseed data')
   519→    } finally {
   520→      setReseeding(false)
   521→    }
   522→  }
   523→
   524→  return (
   525→    <div className="space-y-6">
   526→      {/* Database Info */}
   527→      <Card>
   528→        <CardHeader>
   529→          <CardTitle className="text-lg">Database Overview</CardTitle>
   530→          <CardDescription>Current statistics for your RecruitPro database.</CardDescription>
   531→        </CardHeader>
   532→        <CardContent>
   533→          {statsLoading ? (
   534→            <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   535→              {Array.from({ length: 4 }).map((_, i) => (
   536→                <div key={i} className="h-20 rounded-lg bg-muted animate-pulse" />
   537→              ))}
   538→            </div>
   539→          ) : (
   540→            <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   541→              <div className="rounded-lg border bg-card p-4 text-center">
   542→                <p className="text-2xl font-bold text-primary">{stats.totalCandidates}</p>
   543→                <p className="text-xs text-muted-foreground mt-1">Total Candidates</p>
   544→              </div>
   545→              <div className="rounded-lg border bg-card p-4 text-center">
   546→                <p className="text-2xl font-bold text-emerald-600 dark:text-emerald-400">{stats.totalClients}</p>
   547→                <p className="text-xs text-muted-foreground mt-1">Total Clients</p>
   548→              </div>
   549→              <div className="rounded-lg border bg-card p-4 text-center">
   550→                <p className="text-2xl font-bold text-amber-600 dark:text-amber-400">{stats.totalJobs}</p>
   551→                <p className="text-xs text-muted-foreground mt-1">Total Jobs</p>
   552→              </div>
   553→              <div className="rounded-lg border bg-card p-4 text-center">
   554→                <p className="text-2xl font-bold text-rose-600 dark:text-rose-400">{stats.totalEmployees}</p>
   555→                <p className="text-xs text-muted-foreground mt-1">Total Employees</p>
   556→              </div>
   557→            </div>
   558→          )}
   559→        </CardContent>
   560→      </Card>
   561→
   562→      {/* Actions */}
   563→      <Card>
   564→        <CardHeader>
   565→          <CardTitle className="text-lg">Data Actions</CardTitle>
   566→          <CardDescription>Export, clear, or reseed your application data.</CardDescription>
   567→        </CardHeader>
   568→        <CardContent className="space-y-4">
   569→          <div className="flex flex-col sm:flex-row gap-3">
   570→            <Button
   571→              variant="outline"
   572→              onClick={handleExport}
   573→              disabled={exporting}
   574→              className="flex-1 sm:flex-none"
   575→            >
   576→              {exporting ? (
   577→                <Loader2 className="h-4 w-4 mr-2 animate-spin" />
   578→              ) : (
   579→                <Download className="h-4 w-4 mr-2" />
   580→              )}
   581→              Export All Data (CSV)
